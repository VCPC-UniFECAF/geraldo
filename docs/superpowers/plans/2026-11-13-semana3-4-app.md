**Goal:** Entregar o G.E.R.A.L.D.O. como aplicativo local (`.AppImage` e `.deb`) que confere o ambiente, sobe backend, Gazebo, Nav2, dispatcher e a ponte ROS, serve o `web/` e mostra o robô se mexendo numa visão 2D da simulação.

```python
"""Conversões puras entre mensagens ROS (lidas por atributo) e os dicts da API da ponte. Não importa rclpy."""
import math

LIDAR_OFFSET = (0.275, 0.0)  # lidar_link em relação ao base_link, em metros (geraldo.urdf.xacro)
RANGE_NAMES = ("front_c", "front_l45", "front_r45", "rear_l135", "rear_r135")
MAX_ROUTE_POINTS = 100
MAX_SCAN_POINTS = 360


def yaw_from_quaternion(z: float, w: float) -> float:
    """Yaw (radianos) de uma rotação em torno de Z."""
    return math.atan2(2 * w * z, 1 - 2 * z * z)


def pose_from_amcl(msg) -> dict:
    p = msg.pose.pose
    yaw = yaw_from_quaternion(p.orientation.z, p.orientation.w)
    return {"x": round(p.position.x, 3), "y": round(p.position.y, 3), "yaw": round(yaw, 3)}


def _evenly(items: list, limit: int) -> list:
    """No máximo `limit` itens espaçados por igual, sempre com o primeiro e o último."""
    if len(items) <= limit:
        return list(items)
    step = (len(items) - 1) / (limit - 1)
    return [items[round(i * step)] for i in range(limit)]


def route_from_path(msg, limit: int = MAX_ROUTE_POINTS) -> list[dict]:
    points = [(s.pose.position.x, s.pose.position.y) for s in msg.poses]
    return [{"x": round(x, 3), "y": round(y, 3)} for x, y in _evenly(points, limit)]


def battery_percent(msg) -> float:
    return round(msg.percentage * 100, 1)


def scan_to_dict(msg) -> dict:
    return {
        "angle_min": msg.angle_min,
        "angle_increment": msg.angle_increment,
        "range_min": msg.range_min,
        "range_max": msg.range_max,
        "ranges": list(msg.ranges),
    }


def _hits(scan: dict):
    """(índice, distância) dos feixes que acertaram algo: finitos e dentro do alcance do sensor."""
    for i, r in enumerate(scan["ranges"]):
        if math.isfinite(r) and scan["range_min"] <= r < scan["range_max"]:
            yield i, r


def closest_range(scan: dict) -> float | None:
    """Menor distância medida (metros, 2 casas) ou None se nenhum feixe acertou nada."""
    distances = [r for _, r in _hits(scan)]
    return round(min(distances), 2) if distances else None


def scan_to_map_points(scan: dict, pose: dict, limit: int = MAX_SCAN_POINTS) -> list[dict]:
    """Acertos do LiDAR no referencial `map`, calculados pela pose do AMCL (sem consultar o tf)."""
    cos_y, sin_y = math.cos(pose["yaw"]), math.sin(pose["yaw"])
    ox, oy = LIDAR_OFFSET
    lidar_x = pose["x"] + ox * cos_y - oy * sin_y
    lidar_y = pose["y"] + ox * sin_y + oy * cos_y
    points = []
    for i, r in _hits(scan):
        angle = pose["yaw"] + scan["angle_min"] + i * scan["angle_increment"]
        points.append((lidar_x + r * math.cos(angle), lidar_y + r * math.sin(angle)))
    return [{"x": round(x, 2), "y": round(y, 2)} for x, y in _evenly(points, limit)]
```

```python
import threading
import time
from collections.abc import Callable
from datetime import datetime, timezone

from .convert import RANGE_NAMES, closest_range, scan_to_map_points


class BridgeState:
    """Último valor de cada tópico que a ponte assina. As mensagens chegam na thread do rclpy e o servidor lê em outra."""

    STALE_AFTER_S = 3.0  # sem nenhuma mensagem há mais que isso: a simulação não está falando

    def __init__(
        self,
        clock: Callable[[], float] = time.monotonic,
        now: Callable[[], datetime] = lambda: datetime.now(timezone.utc),
    ) -> None:
        self._lock = threading.Lock()
        self._clock, self._now = clock, now
        self._robot_state: str | None = None
        self._battery: float | None = None
        self._pose: dict | None = None
        self._route: list[dict] = []
        self._scan: list[dict] = []
        self._ranges: dict[str, float | None] = {name: None for name in RANGE_NAMES}
        self._estop = False
        self._last_message: float | None = None

    def _heard(self) -> None:
        self._last_message = self._clock()

    def update_robot_state(self, value: str) -> None:
        with self._lock:
            self._robot_state = value
            self._heard()

    def update_battery(self, percent: float) -> None:
        with self._lock:
            self._battery = percent
            self._heard()

    def update_pose(self, pose: dict) -> None:
        with self._lock:
            self._pose = pose
            self._heard()

    def update_route(self, route: list[dict]) -> None:
        with self._lock:
            self._route = route
            self._heard()

    def update_scan(self, scan: dict) -> None:
        with self._lock:
            self._scan = scan_to_map_points(scan, self._pose) if self._pose else []
            self._heard()

    def update_range(self, name: str, scan: dict) -> None:
        if name not in self._ranges:
            raise ValueError(f"Sensor de alcance desconhecido: {name}. Use um de {', '.join(RANGE_NAMES)}.")
        with self._lock:
            self._ranges[name] = closest_range(scan)
            self._heard()

    def update_estop(self, active: bool) -> None:
        with self._lock:
            self._estop = active

    def snapshot(self) -> dict:
        """Corpo de `GET /snapshot` e de cada mensagem de `WS /ws/sim`."""
        with self._lock:
            stale = self._last_message is None or self._clock() - self._last_message > self.STALE_AFTER_S
            return {
                "robot_state": self._robot_state,
                "pose": dict(self._pose) if self._pose else None,
                "battery": self._battery,
                "route": list(self._route),
                "scan": list(self._scan),
                "ranges": dict(self._ranges),
                "estop": self._estop,
                "stale": stale,
                "updated_at": self._now().isoformat(),
            }
```

```python
import os
import signal
import subprocess
from collections.abc import Callable

GUI_COMMAND = ["gz", "sim", "-g"]


class GuiError(Exception):
    pass


class GazeboGui:
    """Abre a janela 3D do Gazebo (`gz sim -g`) na mesma partição do servidor da simulação. Só uma janela por vez."""

    def __init__(
        self,
        spawn: Callable = subprocess.Popen,
        kill_group: Callable[[int, int], None] = os.killpg,
        partition: str | None = None,
    ) -> None:
        self._spawn = spawn
        self._kill_group = kill_group
        self._partition = partition or os.environ.get("GZ_PARTITION", "verif")
        self._proc = None

    def _running(self) -> bool:
        return self._proc is not None and self._proc.poll() is None

    def open(self) -> tuple[int, bool]:
        """Devolve (pid, já_estava_aberto)."""
        if self._running():
            return self._proc.pid, True
        # no Wayland com GPU híbrida a janela do Gazebo (Ogre2) quebra; roda via X11 (igual ao sim.launch.py)
        env = {**os.environ, "GZ_PARTITION": self._partition, "QT_QPA_PLATFORM": "xcb", "EGL_PLATFORM": ""}
        try:
            self._proc = self._spawn(
                GUI_COMMAND, env=env, start_new_session=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL
            )
        except OSError as err:
            raise GuiError(
                "Não consegui abrir o Gazebo (`gz`). Confira se o Gazebo Harmonic está instalado: bash robot/scripts/check_ros_env.sh."
            ) from err
        return self._proc.pid, False

    def close(self) -> None:
        """Derruba a janela pelo PID do grupo de processos (kill -9); nunca por nome."""
        if self._running():
            self._kill_group(self._proc.pid, signal.SIGKILL)
```

```python
import asyncio

from fastapi import FastAPI, HTTPException, Request, Response, WebSocket, WebSocketDisconnect
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse
from pydantic import BaseModel
from starlette.middleware.trustedhost import TrustedHostMiddleware

from .gazebo_gui import GuiError
from .ports import RosPort
from .state import BridgeState

HOST = "127.0.0.1"  # só a própria máquina: a ponte não tem login
PORT = 8765
ALLOWED_HOSTS = ("127.0.0.1", "localhost")  # barra DNS rebinding
ALLOWED_ORIGINS = ("http://localhost:5173", "http://127.0.0.1:5173")  # o app (e o `npm run dev`) servem o web aqui
WS_INTERVAL_S = 0.1  # 10 mensagens por segundo


class EstopBody(BaseModel):
    active: bool


def create_app(state: BridgeState, port: RosPort, gui, interval_s: float = WS_INTERVAL_S) -> FastAPI:
    app = FastAPI(title="G.E.R.A.L.D.O. ponte ROS", docs_url=None, redoc_url=None, openapi_url=None)

    @app.middleware("http")
    async def reject_foreign_origins(request: Request, call_next):
        origin = request.headers.get("origin")
        if origin is not None and origin not in ALLOWED_ORIGINS:
            return JSONResponse({"detail": "Origem não permitida. Abra a visão 2D pelo aplicativo do G.E.R.A.L.D.O."}, status_code=403)
        return await call_next(request)

    app.add_middleware(
        CORSMiddleware, allow_origins=list(ALLOWED_ORIGINS), allow_methods=["GET", "POST"], allow_headers=["content-type"]
    )
    app.add_middleware(TrustedHostMiddleware, allowed_hosts=list(ALLOWED_HOSTS))

    @app.get("/health")
    def health() -> dict:
        return {"status": "ok"}

    @app.get("/snapshot")
    def snapshot() -> dict:
        return state.snapshot()

    @app.post("/estop", status_code=204, response_class=Response)
    def estop(body: EstopBody) -> None:
        port.publish_estop(body.active)
        state.update_estop(body.active)

    @app.post("/gazebo/gui", status_code=202)
    def open_gazebo() -> dict:
        try:
            pid, already_running = gui.open()
        except GuiError as err:
            raise HTTPException(status_code=500, detail=str(err)) from err
        return {"pid": pid, "already_running": already_running}

    @app.websocket("/ws/sim")
    async def sim_stream(ws: WebSocket) -> None:
        origin = ws.headers.get("origin")
        if origin is not None and origin not in ALLOWED_ORIGINS:
            await ws.close(code=1008)
            return
        await ws.accept()
        try:
            while True:
                await ws.send_json(state.snapshot())
                await asyncio.sleep(interval_s)
        except WebSocketDisconnect:
            pass

    return app
```

```python
import threading

from .convert import (
    RANGE_NAMES,
    battery_percent,
    pose_from_amcl,
    route_from_path,
    scan_to_dict,
)
from .ports import RosPort
from .state import BridgeState

# Contrato (docs/contracts/ros-interfaces.md): são os únicos tópicos que a ponte toca. Nada de comando de movimento.
PUBLISHED = ("/geraldo/estop",)
SUBSCRIBED = (
    "/geraldo/state",
    "/geraldo/battery",
    "/amcl_pose",
    "/plan",
    "/scan",
    *(f"/range/{name}" for name in RANGE_NAMES),
)
ESTOP_REPEATS = (0.0, 0.2, 0.4)  # como `ros2 topic pub -r 5 -t 3`: o dispatcher não perde a mensagem


class RclpyRosPort(RosPort):  # ADAPTER: traduz as mensagens do rclpy para os dicts do BridgeState
    def __init__(self) -> None:
        self._rclpy = None
        self._node = None
        self._executor = None
        self._thread: threading.Thread | None = None
        self._estop_pub = None
        self._bool = None
        self._stopping = threading.Event()

    def start(self, state: BridgeState) -> None:
        import rclpy
        from geometry_msgs.msg import PoseWithCovarianceStamped
        from nav_msgs.msg import Path
        from rclpy.executors import SingleThreadedExecutor
        from rclpy.node import Node
        from rclpy.qos import DurabilityPolicy, QoSProfile, qos_profile_sensor_data
        from sensor_msgs.msg import BatteryState, LaserScan
        from std_msgs.msg import Bool, String

        rclpy.init()
        node = Node("geraldo_bridge")
        # o dispatcher publica o estado "preso" (transient_local); quem entra depois ainda recebe o último
        latched = QoSProfile(depth=1, durability=DurabilityPolicy.TRANSIENT_LOCAL)
        node.create_subscription(String, "/geraldo/state", lambda m: state.update_robot_state(m.data), latched)
        node.create_subscription(BatteryState, "/geraldo/battery", lambda m: state.update_battery(battery_percent(m)), 10)
        node.create_subscription(PoseWithCovarianceStamped, "/amcl_pose", lambda m: state.update_pose(pose_from_amcl(m)), latched)
        node.create_subscription(Path, "/plan", lambda m: state.update_route(route_from_path(m)), 10)
        node.create_subscription(LaserScan, "/scan", lambda m: state.update_scan(scan_to_dict(m)), qos_profile_sensor_data)
        for name in RANGE_NAMES:
            node.create_subscription(
                LaserScan, f"/range/{name}", lambda m, n=name: state.update_range(n, scan_to_dict(m)), qos_profile_sensor_data
            )
        self._estop_pub = node.create_publisher(Bool, "/geraldo/estop", 10)
        self._rclpy, self._node, self._bool = rclpy, node, Bool
        self._executor = SingleThreadedExecutor()
        self._executor.add_node(node)
        self._thread = threading.Thread(target=self._executor.spin, daemon=True)
        self._thread.start()

    def publish_estop(self, active: bool) -> None:
        for delay in ESTOP_REPEATS:
            threading.Timer(delay, self._publish_once, args=(active,)).start()

    def _publish_once(self, active: bool) -> None:
        if self._stopping.is_set() or self._estop_pub is None:  # as repetições ainda pendentes morrem junto com a ponte
            return
        self._estop_pub.publish(self._bool(data=active))

    def stop(self) -> None:
        if self._rclpy is None:
            return
        self._stopping.set()
        self._executor.shutdown()
        self._node.destroy_node()
        if self._rclpy.ok():
            self._rclpy.shutdown()
        self._rclpy = None
```

```python
"""Converte o mapa do Nav2 (hospital.pgm + hospital.yaml) em PNG e JSON para a visão 2D do app.

Uso: python3 robot/scripts/export_map_png.py   (escreve web/public/sim/hospital.png e hospital.json)
"""
import json
import struct
import zlib
from pathlib import Path

import yaml

ROOT = Path(__file__).resolve().parents[2]
MAP_YAML = ROOT / "robot/ros2_ws/src/geraldo_nav/maps/hospital.yaml"
OUT_DIR = ROOT / "web/public/sim"
PNG_NAME = "hospital.png"
JSON_NAME = "hospital.json"


def read_pgm(data: bytes) -> tuple[int, int, bytes]:
    """PGM binário (P5, 255 níveis, sem comentários), como o gen_hospital.py escreve. Devolve (largura, altura, pixels)."""
    try:
        magic, dims, maxval, pixels = data.split(b"\n", 3)
        width, height = (int(n) for n in dims.split())
    except ValueError as err:
        raise ValueError("Mapa ilegível. Gere de novo com python3 robot/hospital/gen_hospital.py.") from err
    if magic != b"P5" or maxval != b"255":
        raise ValueError("Esperava um PGM binário (P5) com 255 níveis. Gere de novo com python3 robot/hospital/gen_hospital.py.")
    if len(pixels) != width * height:
        raise ValueError(f"O PGM declara {width}x{height}, mas tem {len(pixels)} pixels. Gere o mapa de novo.")
    return width, height, pixels


def _chunk(kind: bytes, body: bytes) -> bytes:
    return struct.pack(">I", len(body)) + kind + body + struct.pack(">I", zlib.crc32(kind + body))


def write_png(width: int, height: int, pixels: bytes) -> bytes:
    """PNG em tons de cinza de 8 bits, sem filtro (só a biblioteca padrão)."""
    rows = b"".join(b"\x00" + pixels[y * width : (y + 1) * width] for y in range(height))
    header = struct.pack(">IIBBBBB", width, height, 8, 0, 0, 0, 0)
    return b"\x89PNG\r\n\x1a\n" + _chunk(b"IHDR", header) + _chunk(b"IDAT", zlib.compress(rows, 9)) + _chunk(b"IEND", b"")


def convert(map_yaml: Path = MAP_YAML, out_dir: Path = OUT_DIR) -> None:
    meta = yaml.safe_load(map_yaml.read_text(encoding="utf-8"))
    width, height, pixels = read_pgm((map_yaml.parent / meta["image"]).read_bytes())
    out_dir.mkdir(parents=True, exist_ok=True)
    (out_dir / PNG_NAME).write_bytes(write_png(width, height, pixels))
    info = {
        "image": PNG_NAME,
        "width": width,
        "height": height,
        "resolution": meta["resolution"],
        "origin": meta["origin"],
        "occupied_thresh": meta["occupied_thresh"],
        "free_thresh": meta["free_thresh"],
        "negate": meta["negate"],
    }
    (out_dir / JSON_NAME).write_text(json.dumps(info, indent=2) + "\n", encoding="utf-8")


if __name__ == "__main__":
    convert()
    print(f"gerado: {OUT_DIR / PNG_NAME} e {OUT_DIR / JSON_NAME}")
```

`web/src/sim/MapGrid.ts`:

```ts
import type { MapMeta } from "./types";

export type CellState = "occupied" | "free" | "unknown";

// Mapa de ocupação do Nav2 (tons de cinza do PGM) com a regra de limiares do map_server.
export class MapGrid {
  readonly #meta: MapMeta;
  readonly #gray: ArrayLike<number>;

  constructor(meta: MapMeta, gray: ArrayLike<number>) {
    if (gray.length !== meta.width * meta.height) {
      throw new Error("O mapa tem um número de pixels diferente do declarado. Gere de novo com robot/scripts/export_map_png.py.");
    }
    this.#meta = meta;
    this.#gray = gray;
  }

  get meta(): MapMeta {
    return this.#meta;
  }

  // Estado de um pixel (linha 0 = topo da imagem = y maior do mundo).
  stateOfPixel(col: number, row: number): CellState {
    const { width, height, negate, occupied_thresh, free_thresh } = this.#meta;
    if (col < 0 || row < 0 || col >= width || row >= height) return "unknown";
    const value = this.#gray[row * width + col];
    const occupancy = negate ? value / 255 : (255 - value) / 255;
    if (occupancy > occupied_thresh) return "occupied";
    if (occupancy < free_thresh) return "free";
    return "unknown";
  }

  // Estado da célula que contém o ponto (x, y) do referencial `map`, em metros.
  stateAt(x: number, y: number): CellState {
    const { height, resolution, origin } = this.#meta;
    const col = Math.floor((x - origin[0]) / resolution);
    const row = height - 1 - Math.floor((y - origin[1]) / resolution);
    return this.stateOfPixel(col, row);
  }
}
```

```ts
import type { MapMeta } from "./types";

export interface Viewport {
  scale: number; // pixels por metro
  offsetX: number;
  offsetY: number;
}

// Encaixa o mapa inteiro no canvas, centralizado. O y do mundo cresce para cima; o do canvas, para baixo.
export function fitViewport(meta: MapMeta, width: number, height: number, padding = 8): Viewport {
  const mapW = meta.width * meta.resolution;
  const mapH = meta.height * meta.resolution;
  const scale = Math.max(1e-6, Math.min((width - 2 * padding) / mapW, (height - 2 * padding) / mapH));
  return {
    scale,
    offsetX: (width - mapW * scale) / 2 - meta.origin[0] * scale,
    offsetY: (height + mapH * scale) / 2 + meta.origin[1] * scale,
  };
}

export function worldToCanvas(vp: Viewport, x: number, y: number): [number, number] {
  return [vp.offsetX + x * vp.scale, vp.offsetY - y * vp.scale];
}
```

```ts
export type SimTheme = Record<keyof typeof THEME_TOKENS, string>;

export function readTheme(readVar: (name: string) => string): SimTheme {
  const theme = {} as SimTheme;
  for (const [key, token] of Object.entries(THEME_TOKENS)) {
    const value = readVar(token).trim();
    if (!value) throw new Error(`Falta o token de cor ${token}. Importe design-system/dist/tokens.css na página.`);
    theme[key as keyof SimTheme] = value;
  }
  return theme;

export function themeFromDocument(): SimTheme {
  const style = getComputedStyle(document.documentElement);
  return readTheme((name) => style.getPropertyValue(name));
}

// Aceita os formatos dos tokens: #rgb e #rrggbb.
export function parseCssColor(value: string): [number, number, number] {
  const hex = value.trim().replace(/^#/, "");
  const full = hex.length === 3 ? [...hex].map((c) => c + c).join("") : hex;
  if (!/^[0-9a-fA-F]{6}$/.test(full)) throw new Error(`Cor inválida: "${value}". Use o formato #rrggbb dos tokens.`);
  return [parseInt(full.slice(0, 2), 16), parseInt(full.slice(2, 4), 16), parseInt(full.slice(4, 6), 16)];
}
```

`web/src/sim/rasterize.ts`:

```ts
import type { MapGrid } from "./MapGrid";
import { parseCssColor, type SimTheme } from "./canvasTheme";

// Pinta o mapa com as cores do tema: parede = `parede`, livre = `livre`, desconhecido = ponto médio. Devolve RGBA.
export function rasterize(grid: MapGrid, theme: SimTheme): Uint8ClampedArray {
  const { width, height } = grid.meta;
  const wall = parseCssColor(theme.parede);
  const free = parseCssColor(theme.livre);
  const unknown = wall.map((c, i) => Math.round((c + free[i]) / 2));
  const colors = { occupied: wall, free, unknown };
  const rgba = new Uint8ClampedArray(width * height * 4);
  for (let row = 0; row < height; row++) {
    for (let col = 0; col < width; col++) {
      const [r, g, b] = colors[grid.stateOfPixel(col, row)];
      const at = (row * width + col) * 4;
      rgba[at] = r;
      rgba[at + 1] = g;
      rgba[at + 2] = b;
      rgba[at + 3] = 255;
    }
  }
  return rgba;
}
```

```ts
import type { MapGrid } from "./MapGrid";
import type { Point } from "./types";

export interface ClassifiedScan {
  estaticos: Point[]; // batem com uma parede ou móvel do mapa
  novos: Point[]; // não estão no mapa: gente, maca, carrinho...
}

// Um acerto do LiDAR é "novo" quando não há célula ocupada do mapa a até `tolerance` metros dele
// (a tolerância cobre o erro de localização e a largura do feixe).
export function classifyScan(points: Point[], grid: MapGrid, tolerance = 0.15): ClassifiedScan {
  const step = Math.min(grid.meta.resolution, tolerance);
  const reach = step > 0 ? Math.ceil(tolerance / step) : 0;
  const offsets = Array.from({ length: 2 * reach + 1 }, (_, i) => Math.max(-tolerance, Math.min(tolerance, (i - reach) * step)));
  const nearWall = (p: Point): boolean =>
    offsets.some((dx) => offsets.some((dy) => grid.stateAt(p.x + dx, p.y + dy) === "occupied"));
  const estaticos: Point[] = [];
  const novos: Point[] = [];
  for (const p of points) (nearWall(p) ? estaticos : novos).push(p);
  return { estaticos, novos };
}
```

```ts
import { RANGE_MOUNTS, RANGE_NAMES, FOOTPRINT, type MapMeta, type SimSnapshot } from "./types";
import { worldToCanvas, type Viewport } from "./viewport";

export interface Scene {
  width: number;
  height: number;
  viewport: Viewport;
  meta: MapMeta;
  raster: CanvasImageSource | null; // o mapa já pintado (uma vez por tema)
  snapshot: SimSnapshot | null;
  scan: ClassifiedScan;
  theme: SimTheme;
}

// Desenha a cena inteira. Só usa o contexto 2D e a cena: nada de DOM nem de estado global.
export function drawScene(ctx: CanvasRenderingContext2D, scene: Scene): void {
  const { width, height, viewport: vp, meta, raster, snapshot, scan, theme } = scene;
  ctx.clearRect(0, 0, width, height);

  if (raster) {
    const [left, top] = worldToCanvas(vp, meta.origin[0], meta.origin[1] + meta.height * meta.resolution);
    ctx.imageSmoothingEnabled = false;
    ctx.drawImage(raster, left, top, meta.width * meta.resolution * vp.scale, meta.height * meta.resolution * vp.scale);
  }
  if (!snapshot) return;

  if (snapshot.route.length > 1) {
    ctx.beginPath();
    snapshot.route.forEach((p, i) => {
      const [x, y] = worldToCanvas(vp, p.x, p.y);
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    });
    ctx.strokeStyle = theme.rota;
    ctx.lineWidth = 3;
    ctx.stroke();
  }

  ctx.fillStyle = theme.varredura;
  for (const p of scan.estaticos) {
    const [x, y] = worldToCanvas(vp, p.x, p.y);
    ctx.fillRect(x - 1, y - 1, 2, 2);
  }
  ctx.fillStyle = theme.obstaculo;
  for (const p of scan.novos) {
    const [x, y] = worldToCanvas(vp, p.x, p.y);
    ctx.beginPath();
    ctx.arc(x, y, 3, 0, Math.PI * 2);
    ctx.fill();
  }

  if (snapshot.pose) drawRobot(ctx, scene, snapshot.pose, snapshot);
}

function drawRobot(ctx: CanvasRenderingContext2D, scene: Scene, pose: { x: number; y: number; yaw: number }, snapshot: SimSnapshot): void {
  const { viewport: vp, theme } = scene;
  const [cx, cy] = worldToCanvas(vp, pose.x, pose.y);
  const stopped = snapshot.robot_state === "PARADA_EMERGENCIA";
  const body = stopped ? theme.roboParado : theme.robo;
  const s = vp.scale;
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(-pose.yaw); // yaw cresce no sentido anti-horário; o y do canvas aponta para baixo

  // feixes dos sensores de alcance, do ponto de montagem até a distância medida
  ctx.strokeStyle = theme.alcance;
  ctx.lineWidth = 2;
  for (const name of RANGE_NAMES) {
    const range = snapshot.ranges[name];
    if (range === null) continue;
    const m = RANGE_MOUNTS[name];
    ctx.beginPath();
    ctx.moveTo(m.x * s, -m.y * s);
    ctx.lineTo((m.x + range * Math.cos(m.yaw)) * s, -(m.y + range * Math.sin(m.yaw)) * s);
    ctx.stroke();
  }

  // corpo (retângulo do Nav2) e um bico na frente (+X)
  const halfL = (FOOTPRINT.length / 2) * s;
  const halfW = (FOOTPRINT.width / 2) * s;
  ctx.fillStyle = body;
  ctx.fillRect(-halfL, -halfW, halfL * 2, halfW * 2);
  ctx.beginPath();
  ctx.moveTo(halfL, 0);
  ctx.lineTo(halfL - halfW * 0.6, -halfW * 0.6);
  ctx.lineTo(halfL - halfW * 0.6, halfW * 0.6);
  ctx.closePath();
  ctx.fillStyle = theme.livre;
  ctx.fill();
  ctx.restore();
}
```

```ts
import type { SimSnapshot } from "./types";

export const BRIDGE_URL: string = import.meta.env.VITE_BRIDGE_URL ?? "http://127.0.0.1:8765";

// ISP: a tela precisa de duas coisas separadas, ver a simulação e agir nela.
export interface SimSource {
  // Chama `onSnapshot` a cada mensagem e `onConnection` quando a conexão cai ou volta. Devolve quem cancela.
  subscribe(onSnapshot: (snapshot: SimSnapshot) => void, onConnection: (connected: boolean) => void): () => void;
}

export interface SimActions {
  openGazebo(): Promise<{ pid: number; already_running: boolean }>;
  setEstop(active: boolean): Promise<void>;
}

interface Deps {
  fetchFn?: typeof fetch;
  socketFactory?: (url: string) => WebSocket;
  retryMs?: number;
}

// ADAPTER: traduz a API da ponte ROS (HTTP e WebSocket em 127.0.0.1:8765) para as duas interfaces acima.
export class BridgeSimClient implements SimSource, SimActions {
  readonly #baseUrl: string;
  readonly #fetch: typeof fetch;
  readonly #socketFactory: (url: string) => WebSocket;
  readonly #retryMs: number;

  constructor(baseUrl: string = BRIDGE_URL, deps: Deps = {}) {
    this.#baseUrl = baseUrl;
    this.#fetch = deps.fetchFn ?? ((...args) => fetch(...args));
    this.#socketFactory = deps.socketFactory ?? ((url) => new WebSocket(url));
    this.#retryMs = deps.retryMs ?? 2000;
  }

  subscribe(onSnapshot: (snapshot: SimSnapshot) => void, onConnection: (connected: boolean) => void): () => void {
    const url = `${this.#baseUrl.replace(/^http/, "ws")}/ws/sim`;
    let closed = false;
    let socket: WebSocket | null = null;
    let timer: ReturnType<typeof setTimeout> | undefined;
    const connect = () => {
      const current = this.#socketFactory(url);
      socket = current;
      current.onopen = () => onConnection(true);
      current.onmessage = (event) => onSnapshot(JSON.parse(String(event.data)) as SimSnapshot);
      current.onerror = () => current.close();
      current.onclose = () => {
        onConnection(false);
        if (!closed) timer = setTimeout(connect, this.#retryMs);
      };
    };
    connect();
    return () => {
      closed = true;
      clearTimeout(timer);
      socket?.close();
    };
  }

  async openGazebo(): Promise<{ pid: number; already_running: boolean }> {
    return this.#post("/gazebo/gui");
  }

  async setEstop(active: boolean): Promise<void> {
    await this.#post("/estop", { active });
  }

  async #post<T>(path: string, body?: unknown): Promise<T> {
    let res: Response;
    try {
      res = await this.#fetch(`${this.#baseUrl}${path}`, {
        method: "POST",
        headers: body === undefined ? undefined : { "Content-Type": "application/json" },
        body: body === undefined ? undefined : JSON.stringify(body),
      });
    } catch {
      throw new Error("Não consegui falar com a ponte do robô. Confira se o aplicativo subiu a simulação.");
    }
    if (!res.ok) {
      const detail = (await res.json().catch(() => ({}))) as { detail?: string };
      throw new Error(typeof detail.detail === "string" ? detail.detail : "A ponte recusou o pedido. Tente de novo.");
    }
    return (res.status === 204 ? undefined : await res.json()) as T;
  }
}
```

```ts
import { MapGrid } from "./MapGrid";
import type { MapMeta } from "./types";

// Baixa o PNG e o JSON gerados por robot/scripts/export_map_png.py e devolve o mapa de ocupação.
// Usa o DOM (Image e canvas); a lógica do mapa fica em MapGrid e é testada sem navegador.
export async function loadMap(base = "/sim"): Promise<MapGrid> {
  const res = await fetch(`${base}/hospital.json`);
  if (!res.ok) throw new Error("Não achei o mapa do hospital. Rode python3 robot/scripts/export_map_png.py e construa o web de novo.");
  const meta = (await res.json()) as MapMeta;
  const image = new Image();
  image.src = `${base}/${meta.image}`;
  await image.decode();
  const canvas = document.createElement("canvas");
  canvas.width = meta.width;
  canvas.height = meta.height;
  const ctx = canvas.getContext("2d", { willReadFrequently: true });
  if (!ctx) throw new Error("Este navegador não desenha em canvas 2D.");
  ctx.drawImage(image, 0, 0);
  const rgba = ctx.getImageData(0, 0, meta.width, meta.height).data;
  const gray = new Uint8Array(meta.width * meta.height);
  for (let i = 0; i < gray.length; i++) gray[i] = rgba[i * 4]; // PNG em tons de cinza: R = G = B
  return new MapGrid(meta, gray);
}
```

`app/test/fake_system.mjs` (o `sleep` do falso cede a vez ao laço de eventos com `setImmediate`, para os testes agirem entre uma sondagem e a próxima; o relógio é `clock`, avançado só pelo `sleep`):

```js
// Substituto de teste do `system` (LSP): sem processos, sem disco, sem rede, com relógio controlado.
export class FakeSystem {
  constructor() {
    this.files = new Map();
    this.dirs = new Set();
    this.spawned = []; // {command, args, cwd, env, pid, child}
    this.killed = []; // [pid, signal]
    this.runs = []; // {command, args, opts}
    this.copies = [];
    this.writes = []; // {path, data, mode}
    this.runHandler = () => ({ code: 0, stdout: "", stderr: "" });
    this.httpHandler = () => true;
    this.ps = "";
    this.clock = 0;
    this.nextPid = 1000;
    this.autoExitOnKill = true;
  }

  spawn(command, args, opts = {}) {
    const pid = this.nextPid++;
    const listeners = { exit: [], output: [] };
    const child = {
      pid,
      onExit: (cb) => listeners.exit.push(cb),
      onOutput: (cb) => listeners.output.push(cb),
      emitExit: (code = 0, signal = null) => listeners.exit.forEach((cb) => cb(code, signal)),
      emitOutput: (line, stream = "stdout") => listeners.output.forEach((cb) => cb(line, stream)),
    };
    this.spawned.push({ command, args, cwd: opts.cwd, env: opts.env, pid, child });
    return child;
  }

  async run(command, args, opts = {}) {
    this.runs.push({ command, args, opts });
    return this.runHandler(command, args, opts);
  }

  async httpOk(url) {
    return this.httpHandler(url);
  }

  kill(pid, signal) {
    this.killed.push([pid, signal]);
    if (this.autoExitOnKill) this.spawned.filter((s) => s.pid === pid).forEach((s) => s.child.emitExit(null, signal));
  }

  async psTable() {
    return this.ps;
  }

  async sleep(ms) {
    await new Promise((resolve) => setImmediate(resolve)); // deixa o teste agir entre uma sondagem e a próxima
    this.clock += ms;
  }

  now() {
    return this.clock;
  }

  exists(p) {
    return this.files.has(p) || this.dirs.has(p);
  }

  readFile(p) {
    if (!this.files.has(p)) throw new Error(`ENOENT: ${p}`);
    return this.files.get(p);
  }

  writeFile(p, data, mode) {
    this.files.set(p, data);
    this.writes.push({ path: p, data, mode });
  }

  mkdirp(p) {
    this.dirs.add(p);
  }

  copyDir(from, to) {
    this.copies.push([from, to]);
  }
}
```

```js
export function parsePython(output) {
  const match = /Python (\d+)\.(\d+)/.exec(output);
  return match ? { major: Number(match[1]), minor: Number(match[2]) } : null;
}

// Linhas "FALTA   <o quê>" que o check_ros_env.sh imprime.
export function parseRosCheck(stdout) {
  return stdout
    .split("\n")
    .filter((line) => line.startsWith("FALTA"))
    .map((line) => line.replace(/^FALTA\s+/, "").trim());
}

// O que dizer para cada falta, na ordem em que valem (o primeiro que casar ganha).
const FIXES = [
  [/ROS 2 Jazzy/, "Instale o ROS 2 Jazzy: https://docs.ros.org/en/jazzy/Installation/Ubuntu-Install-Debs.html"],
  [/Gazebo Harmonic/, "Instale o Gazebo Harmonic: sudo apt install ros-jazzy-ros-gz"],
  [/diff-drive/, "Instale o Gazebo do ROS: sudo apt install ros-jazzy-ros-gz ros-jazzy-gz-sim-vendor"],
  [/^nav2/, "Instale o Nav2: sudo apt install ros-jazzy-navigation2 ros-jazzy-nav2-bringup"],
];

export function fixFor(what) {
  const known = FIXES.find(([pattern]) => pattern.test(what));
  if (known) return known[1];
  return `Instale o pacote: sudo apt install ros-jazzy-${what.replaceAll("_", "-")}`;
}

// Nunca lança: devolve {ok, problems: [{what, fix}]}.
export async function checkEnvironment(system, roots) {
  const problems = [];

  const py = await system.run("python3", ["--version"]);
  const version = py.code === 0 ? parsePython(py.stdout + py.stderr) : null;
  if (!version || version.major < 3 || (version.major === 3 && version.minor < 11)) {
    problems.push({ what: "Python 3.11 ou mais novo", fix: "Instale o Python 3: sudo apt install python3 python3-venv" });
  } else {
    const venv = await system.run("python3", ["-c", "import venv, ensurepip"]);
    if (venv.code !== 0) problems.push({ what: "módulo venv do Python", fix: "Instale: sudo apt install python3-venv" });
  }

  const ros = await system.run("bash", [path.join(roots.scriptsDir, "check_ros_env.sh")]);
  if (ros.code !== 0) {
    const missing = parseRosCheck(ros.stdout);
    if (missing.length === 0) {
      problems.push({ what: "o conferidor do ambiente ROS falhou", fix: `Rode bash ${path.join(roots.scriptsDir, "check_ros_env.sh")} e leia a mensagem: ${(ros.stderr || ros.stdout).trim()}` });
    }
    for (const what of missing) problems.push({ what, fix: fixFor(what) });
  } else {
    const colcon = await system.run("bash", ["-c", `source ${shq(ROS_SETUP)} && command -v colcon`]);
    if (colcon.code !== 0) problems.push({ what: "colcon", fix: "Instale: sudo apt install python3-colcon-common-extensions" });
  }

  return { ok: problems.length === 0, problems };
}
```

```js
import { createHash } from "node:crypto";
import path from "node:path";
import { ROS_SETUP, installSetup, requirementsFiles } from "./config.mjs";
import { shq } from "./shell.mjs";

export class PrepareError extends Error {}

async function mustRun(system, command, args, what, opts = {}) {
  const res = await system.run(command, args, { timeoutMs: 15 * 60 * 1000, ...opts });
  if (res.code !== 0) {
    const tail = (res.stderr || res.stdout).trim().split("\n").slice(-5).join("\n");
    throw new PrepareError(`Não consegui ${what}. Confira a internet e o espaço em disco e abra o app de novo.\n${tail}`);
  }
}

// venv do backend e da ponte: criado na primeira execução e refeito só quando os requirements mudam.
// `--system-site-packages` é obrigatório: as mensagens do ROS (rclpy, geometry_msgs) precisam do numpy do sistema.
export async function ensureVenv(system, roots, onProgress = () => {}) {
  const files = requirementsFiles(roots);
  const missing = files.find((f) => !system.exists(f));
  if (missing) throw new PrepareError(`Falta o arquivo ${missing}. Reinstale o aplicativo ou, no repositório, confira se o backend e a ponte têm requirements.txt.`);
  const wanted = createHash("sha256").update(files.map((f) => system.readFile(f)).join("\n--\n")).digest("hex");
  const stampFile = path.join(roots.venvDir, ".requirements.sha256");
  if (system.exists(roots.venvPython) && system.exists(stampFile) && system.readFile(stampFile).trim() === wanted) return false;

  if (!system.exists(roots.venvPython)) {
    onProgress("Criando o ambiente Python (primeira execução)…");
    system.mkdirp(roots.dataDir);
    await mustRun(system, "python3", ["-m", "venv", "--system-site-packages", roots.venvDir], "criar o ambiente Python");
  }
  onProgress("Instalando as dependências do backend e da ponte…");
  const requirements = files.flatMap((f) => ["-r", f]);
  await mustRun(system, roots.venvPython, ["-m", "pip", "install", "--disable-pip-version-check", ...requirements], "instalar as dependências Python");
  system.writeFile(stampFile, wanted + "\n");
  return true;
}

// Compila os pacotes ROS. Em desenvolvimento só se ainda não existe `install/` (depois de mudar um pacote, rode o colcon).
// Instalado, copia os fontes para o diretório de dados e recompila quando a versão do app muda (`stamp`).
export async function ensureWorkspace(system, roots, stamp, onProgress = () => {}) {
  const stampFile = path.join(roots.wsDir, ".geraldo-stamp");
  const built = system.exists(installSetup(roots));
  const current = stamp === null || (system.exists(stampFile) && system.readFile(stampFile).trim() === stamp);
  if (built && current) return false;

  if (roots.packaged) {
    onProgress("Copiando os pacotes ROS…");
    system.copyDir(roots.wsSrcDir, path.join(roots.wsDir, "src"));
  }
  onProgress("Compilando os pacotes ROS (primeira execução, leva alguns minutos)…");
  await mustRun(
    system,
    "bash",
    ["-c", `source ${shq(ROS_SETUP)} && cd ${shq(roots.wsDir)} && colcon build --symlink-install`],
    "compilar os pacotes ROS",
    { timeoutMs: 30 * 60 * 1000 },
  );
  if (stamp !== null) system.writeFile(stampFile, stamp + "\n");
  return true;
}
```

```js
// Processos filhos por PID. Nunca por nome ou padrão de linha de comando: isso derruba o próprio shell
// e pode pegar a simulação de outra sessão.

// `ps -eo pid=,ppid=` -> PIDs do processo e de todos os descendentes, folhas primeiro.
export function descendantsOf(psTable, rootPid) {
  const children = new Map();
  for (const line of psTable.split("\n")) {
    const [pid, ppid] = line.trim().split(/\s+/).map(Number);
    if (!Number.isInteger(pid) || !Number.isInteger(ppid)) continue;
    if (!children.has(ppid)) children.set(ppid, []);
    children.get(ppid).push(pid);
  }
  const order = [];
  const visit = (pid, seen) => {
    if (seen.has(pid)) return;
    seen.add(pid);
    for (const child of children.get(pid) ?? []) visit(child, seen);
    order.push(pid);
  };
  visit(rootPid, new Set());
  return order;
}

// Manda o sinal ao processo e a todos os descendentes (o `gz sim` e o `ros2 launch` deixam filhos).
export async function killTree(system, rootPid, signal = "SIGKILL") {
  if (!Number.isInteger(rootPid) || rootPid < 2) return [];
  const pids = descendantsOf(await system.psTable(), rootPid).filter((pid) => pid >= 2);
  for (const pid of pids) system.kill(pid, signal);
  return pids;
}
```

```js
import { killTree } from "./process_tree.mjs";

const TERM_GRACE_MS = 5000;

export class ProcessError extends Error {
  constructor(step, message) {
    super(message);
    this.step = step; // qual etapa do stack falhou
  }
}

// Sobe os processos na ordem das dependências (os independentes em paralelo), espera cada um ficar pronto
// e os derruba ao contrário. Fala com o mundo só pelo `system` injetado.
export class Supervisor {
  #system;
  #specs;
  #emit;
  #onLog;
  #pollMs;
  #running = new Map();
  #stopping = false;

  constructor({ system, specs, onEvent = () => {}, onLog = () => {}, pollMs = 1000 }) {
    this.#system = system;
    this.#specs = specs;
    this.#emit = onEvent;
    this.#onLog = onLog;
    this.#pollMs = pollMs;
  }

  get pids() {
    return Object.fromEntries([...this.#running].filter(([, e]) => !e.exited).map(([name, e]) => [name, e.child.pid]));
  }

  async start() {
    const ready = new Map();
    for (const spec of this.#specs) {
      ready.set(spec.name, Promise.all(spec.dependsOn.map((dep) => ready.get(dep))).then(() => this.#startProcess(spec)));
    }
    try {
      await Promise.all(ready.values());
    } catch (err) {
      await this.stop();
      throw err;
    }
  }

  async #startProcess(spec) {
    if (this.#stopping) throw new Error("Parada pedida durante a inicialização.");
    this.#emit({ step: spec.name, state: "running", detail: "Iniciando…" });
    const child = this.#system.spawn(spec.command, spec.args, { cwd: spec.cwd, env: spec.env });
    const entry = { child, spec, exited: false };
    this.#running.set(spec.name, entry);
    child.onOutput((line, stream) => this.#onLog(spec.name, line, stream));
    child.onExit((code, signal) => {
      if (entry.exited) return;
      entry.exited = true;
      if (!this.#stopping) this.#emit({ step: spec.name, state: "failed", detail: `Terminou sozinho (${code ?? signal}). Veja o log em ${spec.name}.log.` });
    });
    await this.#waitReady(spec);
    this.#emit({ step: spec.name, state: "ok", detail: `PID ${child.pid}` });
  }

  async #waitReady(spec) {
    const system = this.#system;
    const { ready } = spec;
    if (ready.kind === "none") return;
    const deadline = system.now() + ready.timeoutMs;
    for (;;) {
      if (this.#stopping) throw new Error("Parada pedida durante a inicialização.");
      if (this.#running.get(spec.name).exited) throw new ProcessError(spec.name, `${spec.name} terminou antes de ficar pronto. Veja o log em ${spec.name}.log.`);
      if (await this.#probe(spec)) return;
      if (system.now() >= deadline) {
        throw new ProcessError(spec.name, `${spec.name} não ficou pronto em ${Math.round(ready.timeoutMs / 1000)} s. Veja o log em ${spec.name}.log e tente de novo.`);
      }
      await system.sleep(this.#pollMs);
    }
  }

  async #probe({ ready, env, cwd }) {
    if (ready.kind === "http") return this.#system.httpOk(ready.url);
    const result = await this.#system.run(ready.command, ready.args, { env, cwd, timeoutMs: 15_000 });
    return result.code === 0;
  }

  async stop() {
    this.#stopping = true;
    for (const entry of [...this.#running.values()].reverse()) await this.#stopOne(entry);
    this.#running.clear();
  }

  async #stopOne(entry) {
    const { spec, child } = entry;
    if (!entry.exited) {
      if (spec.stop === "kill9") {
        await killTree(this.#system, child.pid, "SIGKILL");
      } else {
        this.#system.kill(child.pid, "SIGTERM");
        const deadline = this.#system.now() + TERM_GRACE_MS;
        while (!entry.exited && this.#system.now() < deadline) await this.#system.sleep(100);
        if (!entry.exited) await killTree(this.#system, child.pid, "SIGKILL");
      }
    }
    entry.exited = true;
    this.#emit({ step: spec.name, state: "stopped" });
  }
}
```

```js
  #roots;

  #demoPassword;

      this.#emit({ step, state: "ok", detail });

      throw err instanceof StartError || err instanceof PrepareError ? err : new StartError(err.message);

    await this.#step("venv", async () => ((await ensureVenv(system, roots, progress("venv"))) ? "Ambiente Python criado" : "Ambiente Python em dia"));

    try {
      await this.#supervisor.start();
    } catch (err) {
      if (err.step) this.#emit({ step: err.step, state: "failed", detail: err.message });
      throw new StartError(err.message);
    }
    this.#emit({ step: "pronto", state: "ok", detail: "Tudo no ar: o sistema, a simulação e a ponte estão rodando." });
  }

    await this.#supervisor?.stop();
```

```js
import { execFile, spawn } from "node:child_process";
import { cpSync, existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";

function lines(onLine) {
  let rest = "";
  return (chunk) => {
    const parts = (rest + chunk.toString()).split("\n");
    rest = parts.pop();
    parts.forEach(onLine);
  };
}

// ADAPTER: o `system` que o resto do código usa, sobre child_process, fetch e fs. Nos testes entra o FakeSystem.
export function createNodeSystem() {
  return {
    spawn(command, args, { cwd, env } = {}) {
      const proc = spawn(command, args, { cwd, env: { ...process.env, ...env }, stdio: ["ignore", "pipe", "pipe"] });
      return {
        pid: proc.pid,
        onExit(cb) {
          proc.on("exit", (code, signal) => cb(code, signal));
          proc.on("error", (err) => cb(null, err.code ?? "ERRO")); // por exemplo ENOENT: o comando não existe
        },
        onOutput(cb) {
          proc.stdout.on("data", lines((line) => cb(line, "stdout")));
          proc.stderr.on("data", lines((line) => cb(line, "stderr")));
        },
      };
    },

    run(command, args, { cwd, env, timeoutMs = 60_000 } = {}) {
      return new Promise((resolve) => {
        execFile(
          command,
          args,
          { cwd, env: { ...process.env, ...env }, timeout: timeoutMs, maxBuffer: 16 * 1024 * 1024 },
          (error, stdout, stderr) => resolve({ code: error ? (typeof error.code === "number" ? error.code : 127) : 0, stdout, stderr }),
        );
      });
    },

    async httpOk(url) {
      try {
        return (await fetch(url, { signal: AbortSignal.timeout(2000) })).ok;
      } catch {
        return false;
      }
    },

    kill(pid, signal) {
      try {
        process.kill(pid, signal);
      } catch (err) {
        if (err.code !== "ESRCH") throw err; // já tinha saído
      }
    },

    async psTable() {
      return (await this.run("ps", ["-eo", "pid=,ppid="])).stdout;
    },

    sleep: (ms) => new Promise((resolve) => setTimeout(resolve, ms)),
    now: () => Date.now(),
    exists: (path) => existsSync(path),
    readFile: (path) => readFileSync(path, "utf8"),
    writeFile: (path, data, mode) => writeFileSync(path, data, mode === undefined ? undefined : { mode }),
    mkdirp: (path) => mkdirSync(path, { recursive: true }),
    copyDir: (from, to) => cpSync(from, to, { recursive: true, force: true }),
```

```js
import { readFile } from "node:fs/promises";
import http from "node:http";
import path from "node:path";

const MIME = {
  ".html": "text/html; charset=utf-8",
  ".js": "text/javascript; charset=utf-8",
  ".mjs": "text/javascript; charset=utf-8",
  ".css": "text/css; charset=utf-8",
  ".json": "application/json; charset=utf-8",
  ".png": "image/png",
  ".svg": "image/svg+xml",
  ".ico": "image/x-icon",
  ".woff2": "font/woff2",
  ".map": "application/json",
};

// Pura: qual arquivo atende `pathname`, ou null (fora das pastas montadas, com `..`, com bytes inválidos).
// mounts = [{ prefix: "/__app/", dir }, { prefix: "/", dir }]; o prefixo mais longo ganha.
export function resolveFile(mounts, pathname) {
  let decoded;
  try {
    decoded = decodeURIComponent(pathname);
  } catch {
    return null;
  }
  if (decoded.includes("\0")) return null;
  const sorted = [...mounts].sort((a, b) => b.prefix.length - a.prefix.length);
  const mount = sorted.find((m) => decoded.startsWith(m.prefix));
  if (!mount) return null;
  let rel = decoded.slice(mount.prefix.length);
  if (rel === "" || rel.endsWith("/")) rel += "index.html";
  const root = path.resolve(mount.dir);
  const full = path.resolve(root, rel);
  return full.startsWith(root + path.sep) ? full : null;
}

// Serve o web construído (e a tela de status do app) só em 127.0.0.1. Só GET e HEAD.
export function createStaticServer({ mounts, port, host = "127.0.0.1" }) {
  const server = http.createServer(async (req, res) => {
    const headers = { "X-Content-Type-Options": "nosniff", "Cache-Control": "no-store" };
    if (req.method !== "GET" && req.method !== "HEAD") {
      res.writeHead(405, { ...headers, Allow: "GET, HEAD" }).end();
      return;
    }
    const file = resolveFile(mounts, new URL(req.url, "http://localhost").pathname);
    try {
      if (!file) throw Object.assign(new Error("fora"), { code: "ENOENT" });
      const body = await readFile(file);
      res.writeHead(200, { ...headers, "Content-Type": MIME[path.extname(file)] ?? "application/octet-stream" });
      res.end(req.method === "HEAD" ? undefined : body);
    } catch {
      res.writeHead(404, { ...headers, "Content-Type": "text/plain; charset=utf-8" }).end("Não encontrado.");
    }
  });
  return new Promise((resolve, reject) => {
    server.once("error", (err) =>
      reject(
        err.code === "EADDRINUSE"
          ? new Error(`A porta ${port} está ocupada. Feche o que a usa (por exemplo, o npm run dev do web) e abra o app de novo.`)
          : err,
      ),
    );
    server.listen(port, host, () => resolve({ port: server.address().port, close: () => new Promise((done) => server.close(done)) }));
  });
}
```

```js
import { createRuntime } from "./runtime.mjs";
```

```js
// O AppImage não instala o chrome-sandbox com SUID e o Ubuntu 24.04 restringe user namespaces: sem isto a janela não abre.
// O app só carrega páginas servidas por ele mesmo, em localhost. O .deb mantém o sandbox.

let runtime = null;
let model = initialModel();
let statusWindow = null;
let quitting = false;
const pages = { sistema: null, simulacao: null };

function createWindow(pathname, options) {
  const win = new BrowserWindow(windowOptions(options));
  win.webContents.setWindowOpenHandler(() => ({ action: "deny" }));
  win.webContents.on("will-navigate", (event, url) => {
    if (!isAllowedUrl(url, ORIGIN)) event.preventDefault();
  });
  win.loadURL(ORIGIN + pathname);
  return win;
}

function showPage(key, pathname, title) {
  const open = pages[key];
  if (open && !open.isDestroyed()) return open.focus();
  pages[key] = createWindow(pathname, { title });
}

function pushStatus() {
  if (statusWindow && !statusWindow.isDestroyed()) statusWindow.webContents.send("geraldo:status", model);
}

async function boot() {
  model = initialModel();
  pushStatus();
  try {
    await runtime.startStack();
  } catch {
    return; // o evento de falha já levou a mensagem para a tela de status
  }
  showPage("sistema", "/", "G.E.R.A.L.D.O.");
  showPage("simulacao", "/sim.html", "G.E.R.A.L.D.O. — Simulação 2D");
  statusWindow?.close(); // depois de abrir as outras, para o app não achar que acabou
}

function fromStatusPage(event) {
  return isStatusPage(event.senderFrame?.url ?? "", ORIGIN);
}

async function main() {
  runtime = createRuntime({
    packaged: app.isPackaged,
    appDir: path.resolve(here, ".."),
    resourcesPath: process.resourcesPath,
    appVersion: app.getVersion(),
    onEvent: (event) => {
      model = applyEvent(model, event);
      pushStatus();
    },
  });
  try {
    await runtime.serve();
  } catch (err) {
    dialog.showErrorBox("G.E.R.A.L.D.O.", err.message);
    return app.quit();
  }

  Menu.setApplicationMenu(
    Menu.buildFromTemplate([
      {
        label: "Janelas",
        submenu: [
          { label: "Sistema", accelerator: "CmdOrCtrl+1", click: () => showPage("sistema", "/", "G.E.R.A.L.D.O.") },
          { label: "Simulação 2D", accelerator: "CmdOrCtrl+2", click: () => showPage("simulacao", "/sim.html", "G.E.R.A.L.D.O. — Simulação 2D") },
          { type: "separator" },
          { role: "quit", label: "Sair" },
        ],
      },
    ]),
  );
  ipcMain.on("geraldo:retry", async (event) => {
    if (!fromStatusPage(event)) return;
    await runtime.stopStack();
    boot();
  });
  ipcMain.on("geraldo:quit", (event) => {
    if (fromStatusPage(event)) app.quit();
  });

  statusWindow = createWindow("/__app/status.html", {
    title: "G.E.R.A.L.D.O.",
    width: 640,
    height: 600,
    preload: path.join(here, "preload.cjs"),
  });
  statusWindow.webContents.on("did-finish-load", pushStatus);
  boot();
}

app.on("window-all-closed", () => app.quit());
app.on("before-quit", (event) => {
  if (quitting || !runtime) return;
  quitting = true;
  event.preventDefault();
  runtime.stop().finally(() => app.exit(0)); // derruba backend, Gazebo, Nav2, despachante e ponte por PID
});

if (app.requestSingleInstanceLock()) app.whenReady().then(main);
else app.quit(); // uma segunda cópia brigaria pelas portas 5173, 8000 e 8765
```

```js
// Tela de status: desenha o modelo que o processo principal envia. Só textContent: nada de HTML vindo de log.
const LABELS = { pending: "Aguardando", running: "Em andamento", ok: "Pronto", failed: "Falhou", stopped: "Parado" };
const MARKS = { pending: "…", running: "⟳", ok: "✓", failed: "✗", stopped: "■" };

function render(model) {
  const list = document.getElementById("steps");
  list.replaceChildren(
    ...model.steps.map((step) => {
      const item = document.createElement("li");
      item.dataset.state = step.state;
      const title = document.createElement("span");
      title.textContent = `${step.label}: `;
      const state = document.createElement("span");
      state.className = "state";
      state.textContent = `${MARKS[step.state]} ${LABELS[step.state]}`;
      item.append(title, state);
      if (step.detail) {
        const detail = document.createElement("span");
        detail.className = "detail";
        detail.textContent = step.detail;
        item.append(detail);
      }
      return item;
    }),
  );
  const message = document.getElementById("message");
  message.hidden = !model.failed;
  message.textContent = model.failed ? model.message : "";
  document.getElementById("retry").hidden = !model.failed;
  document.getElementById("summary").textContent = model.done
    ? "Tudo no ar. Abrindo o sistema e a simulação."
    : "Preparando o sistema e a simulação. Isso leva alguns minutos na primeira vez.";
}

window.geraldo.onStatus(render);
document.getElementById("retry").addEventListener("click", () => window.geraldo.retry());
document.getElementById("quit").addEventListener("click", () => window.geraldo.quit());
```
