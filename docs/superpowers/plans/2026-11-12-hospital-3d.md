# Hospital 3D (maquete) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Trocar o hospital de caixas do Gazebo por uma maquete 3D (materiais, mobiliário, obstáculos e maca spawnável) gerada por script no Blender a partir do `hospital.yaml`.

**Architecture:** `hospital.yaml` (fonte única) ganha `furniture`, `obstacles` e `stretcher`. `gen_hospital.py` valida as peças contra o catálogo `pieces.py`, põe os obstáculos no mapa do Nav2 e escreve `hospital_layout.json` (o Python do Blender não tem PyYAML). `modelos-3d/build_hospital.py` lê esse JSON e monta o `hospital.blend`; `export_hospital_mesh.py` exporta OBJ/MTL que o SDF do mundo referencia por `package://`.

**Tech Stack:** Python 3 (pytest, PyYAML, numpy), Blender 5.2 (`bpy`, `bmesh`), Gazebo Harmonic, ROS 2 Jazzy, Nav2.

**Spec:** `docs/superpowers/specs/2026-11-12-hospital-3d-design.md`

## Global Constraints

- Unidades em metros; origem no canto sudoeste, x leste, y norte, z para cima.
- `hospital.yaml` é a fonte única; `worlds/hospital.sdf`, `maps/hospital.*`, `waypoints.yaml`, `locais.json`, `hospital_layout.json` e `worlds/maca.sdf` são **gerados**: nunca editar à mão.
- Mobiliário só visual; só `obstacles` (e a maca spawnada) têm colisão. Os 10 pontos de entrega continuam alcançáveis e existe rota alternativa com a maca.
- Faixa livre de 1,2 m entre cada porta e o ponto de parada da sala (`LANE_WIDTH = 1.2`).
- Cores da maquete só no dicionário `PALETA` (`modelos-3d/hospital_paleta.py`).
- Sem teto; materiais PBR simples, sem texturas de imagem; ~500 mil triângulos no máximo.
- Tipo de peça desconhecido falha em português: `peça desconhecida 'X'. Tipos válidos: ...`.
- Nenhum contrato muda (sem tópico ROS, endpoint ou campo novo).
- Todo código novo vem com teste (ver, falhar, implementar, ver passar). Commits pequenos em português, `tipo: o que mudou`, terminando com a linha `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Comandos de shell com prefixo `rtk` quando a saída for lida crua.

```python
# ---------- peças (mobiliário, obstáculos, maca) ----------
Rect = tuple[float, float, float, float]
LANE_WIDTH = 1.2


def _placed(item: dict, key: str, group: str | None, default_room: str) -> dict:
    kind = item["type"]
    if kind not in PIECES:
        raise ValueError(f"peça desconhecida '{kind}'. Tipos válidos: {', '.join(sorted(PIECES))}")
    piece = PIECES[kind]
    if group and piece.group != group:
        raise ValueError(f"peça '{kind}' é do grupo '{piece.group}' e não pode ficar em '{group}'")
    yaw = item.get("yaw", 0)
    if yaw % 90:
        raise ValueError(f"peça '{kind}': yaw {yaw} precisa ser múltiplo de 90")
    w, d, h = piece.size
    if yaw % 180:
        w, d = d, w
    x, y = item["xy"]
    return {
        "id": key, "type": kind, "room": item.get("room", default_room), "xy": [x, y], "yaw": yaw,
        "size": list(piece.size), "z": piece.z, "group": piece.group, "walkable": piece.walkable,
        "rect": [x - w / 2, y - d / 2, x + w / 2, y + d / 2],
    }


def placed_pieces(spec: dict) -> list[dict]:
    """Todas as peças do YAML (mobiliário, obstáculos e a maca) com tamanho, grupo e retângulo no chão."""
    count: dict[str, int] = {}
    out = []
    for section, group in (("furniture", MOBILIARIO), ("obstacles", OBSTACULO)):
        for item in spec.get(section, []):
            n = count[item["type"]] = count.get(item["type"], 0) + 1
            out.append(_placed(item, f"{item['type']}_{n}", group, "corredor"))
    if "stretcher" in spec:
        stretcher = _placed(spec["stretcher"], "maca", None, "corredor")
        stretcher["group"] = "maca"
        out.append(stretcher)
    return out


def obstacle_rects(spec: dict) -> list[Rect]:
    return [tuple(p["rect"]) for p in placed_pieces(spec) if p["group"] == OBSTACULO]


def stretcher_rect(spec: dict) -> Rect:
    return tuple(next(p for p in placed_pieces(spec) if p["group"] == "maca")["rect"])


def _door_points(room: dict) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = room["rect"]
    return [{"N": (d["at"], y1), "S": (d["at"], y0), "W": (x0, d["at"]), "E": (x1, d["at"])}[d["wall"]] for d in room["doors"]]


def lanes(spec: dict) -> dict[str, list[Rect]]:
    """Faixas que ficam livres de móveis: de cada porta até o ponto de parada (ou o centro da sala) e dos outros pontos até ele."""
    r = LANE_WIDTH / 2
    out = {}
    for room in spec["rooms"]:
        x0, y0, x1, y1 = room["rect"]
        stops = {wp["id"]: wp["xy"] for wp in room["waypoints"]}
        target = stops[room["stop"]] if room.get("stop") else [(x0 + x1) / 2, (y0 + y1) / 2]
        starts = _door_points(room) + [xy for xy in stops.values() if xy != target]
        out[room["id"]] = [
            (min(a[0], target[0]) - r, min(a[1], target[1]) - r, max(a[0], target[0]) + r, max(a[1], target[1]) + r)
            for a in starts
        ]
    return out
```

def door_centers(spec: dict) -> list[dict]:
    """Centro de cada vão (portas das salas e entradas); `axis` é a direção em que a parede corre."""
    doors = []
    for room in spec["rooms"]:
        for (x, y), d in zip(_door_points(room), room["doors"]):
            doors.append({"x": float(x), "y": float(y), "axis": "x" if d["wall"] in ("N", "S") else "y"})
    w, h = spec["size"]
    for e in spec["entrances"]:
        x, y = {"N": (e["at"], h), "S": (e["at"], 0), "W": (0, e["at"]), "E": (w, e["at"])}[e["wall"]]
        doors.append({"x": float(x), "y": float(y), "axis": "x" if e["wall"] in ("N", "S") else "y"})
    return doors


def build_layout(spec: dict) -> dict:
    """Tudo que o Blender precisa (ele não tem PyYAML): paredes, vãos, salas e peças."""
    return {
        "size": spec["size"],
        "wall_thickness": spec["wall_thickness"],
        "wall_height": spec["wall_height"],
        "door_width": spec["door_width"],
        "walls": [list(s) for s in wall_segments(spec)],
        "doors": door_centers(spec),
        "rooms": [{"id": r["id"], "kind": r["kind"], "rect": r["rect"]} for r in spec["rooms"]],
        "pieces": placed_pieces(spec),
    }


def write_layout(spec: dict, path: Path) -> None:
    path.write_text(json.dumps(build_layout(spec), ensure_ascii=False, indent=1) + "\n", encoding="utf-8")

```python
"""Biblioteca de peças do hospital. Cada função registrada com @peca("tipo") desenha a peça dentro da caixa (w, d, h):
centrada em x e y, com o pé em z = 0. Os tamanhos vêm de robot/hospital/pieces.py (fonte única)."""
import math

import bmesh
import bpy
from mathutils import Matrix, Vector

from hospital_paleta import PALETA

PECAS = {}


def peca(tipo):
    def registra(fn):
        PECAS[tipo] = fn
        return fn
    return registra


def material(nome):
    m = bpy.data.materials.get(nome)
    if m:
        return m
    cor, rugosidade, metal = PALETA[nome]
    m = bpy.data.materials.new(nome)
    try:
        m.use_nodes = True
    except Exception:
        pass
    b = m.node_tree.nodes["Principled BSDF"]
    b.inputs["Base Color"].default_value = (*cor, 1)
    b.inputs["Roughness"].default_value = rugosidade
    b.inputs["Metallic"].default_value = metal
    m.diffuse_color = (*cor, 1)
    return m


class Forma:
    """Acumula caixas e cilindros (coordenadas locais) numa malha só, com um slot de material por nome da PALETA."""

    def __init__(self):
        self.bm = bmesh.new()
        self.slots = []

    def _atribui(self, verts, mat):
        if mat not in self.slots:
            self.slots.append(mat)
        idx = self.slots.index(mat)
        for face in {f for v in verts for f in v.link_faces}:
            face.material_index = idx

    def caixa(self, tam, pos, mat):
        """Caixa de tamanho `tam` (x, y, z) com o centro em `pos`."""
        verts = bmesh.ops.create_cube(self.bm, size=1.0)["verts"]
        bmesh.ops.scale(self.bm, vec=Vector(tam), verts=verts)
        bmesh.ops.translate(self.bm, vec=Vector(pos), verts=verts)
        self._atribui(verts, mat)

    def cilindro(self, raio, alt, pos, mat, eixo="z", seg=16):
        """Cilindro de raio `raio` e altura `alt` ao longo de `eixo`, com o centro em `pos`."""
        verts = bmesh.ops.create_cone(self.bm, cap_ends=True, segments=seg, radius1=raio, radius2=raio, depth=alt)["verts"]
        if eixo != "z":
            giro = Matrix.Rotation(math.pi / 2, 3, "Y" if eixo == "x" else "X")
            bmesh.ops.rotate(self.bm, cent=(0, 0, 0), matrix=giro, verts=verts)
        bmesh.ops.translate(self.bm, vec=Vector(pos), verts=verts)
        self._atribui(verts, mat)

    def objeto(self, nome, colecao, loc=(0, 0, 0), yaw=0.0):
        me = bpy.data.meshes.new(nome)
        self.bm.to_mesh(me)
        self.bm.free()
        for nome_mat in self.slots:
            me.materials.append(material(nome_mat))
        obj = bpy.data.objects.new(nome, me)
        obj.location = loc
        obj.rotation_euler = (0, 0, math.radians(yaw))
        colecao.objects.link(obj)
        return obj
```

def estrutura(layout, col):
    """Piso (vinílico + uma placa por sala), paredes com rodapé e faixa de proteção, batentes e vergas das portas."""
    t, h, dw = layout["wall_thickness"], layout["wall_height"], layout["door_width"]
    w, d = layout["size"]
    f = Forma()
    f.caixa((w, d, 0.006), (w / 2, d / 2, 0.003), "piso_vinilico")
    for r in layout["rooms"]:
        x0, y0, x1, y1 = r["rect"]
        f.caixa((x1 - x0, y1 - y0, 0.008), ((x0 + x1) / 2, (y0 + y1) / 2, 0.004), PISO.get(r["kind"], "piso_porcelanato"))

    for x0, y0, x1, y1 in layout["walls"]:
        chave = tuple(round(v, 3) for v in (min(x0, x1), min(y0, y1), max(x0, x1), max(y0, y1)))
        if chave in vistas or abs(x1 - x0) + abs(y1 - y0) < 1e-6:
            continue
        vistas.add(chave)
        horizontal = abs(y1 - y0) < 1e-9
        comp = abs(x1 - x0) if horizontal else abs(y1 - y0)
        sx, sy = (comp + t, t) if horizontal else (t, comp + t)
        cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
        f.caixa((sx, sy, h), (cx, cy, h / 2), "parede")
        f.caixa((sx + 0.01, sy + 0.01, 0.12), (cx, cy, 0.06), "rodape")
        f.caixa((sx + 0.01, sy + 0.01, 0.25), (cx, cy, 0.9), "faixa_protecao")
    for dr in layout["doors"]:
        x, y = dr["x"], dr["y"]
        along = dr["axis"] == "x"
        for s in (-1, 1):     # batentes: a parede termina exatamente na borda do vão
            off = s * (dw / 2 + 0.03)
            f.caixa((0.06, t + 0.04, 2.1) if along else (t + 0.04, 0.06, 2.1), (x + off, y, 1.05) if along else (x, y + off, 1.05), "laca_branca")
        f.caixa((dw + 0.12, t + 0.04, 0.4) if along else (t + 0.04, dw + 0.12, 0.4), (x, y, 2.3), "laca_branca")
    return f.objeto("estrutura", col)

def renderizar(vistas):

```python
"""Exporta as 4 malhas do hospital para o Gazebo (OBJ + MTL).

Uso: blender -b modelos-3d/hospital.blend --python robot/scripts/export_hospital_mesh.py [-- pasta_de_saida]
Sem pasta de saída, grava em robot/ros2_ws/src/geraldo_description/meshes/hospital.

hospital_estrutura, hospital_mobiliario e hospital_obstaculos saem em coordenadas do mundo (origem no canto sudoeste).
A maca sai na origem e sem giro, porque o maca.sdf a posiciona (é spawnada em tempo de execução).
Eixos: forward="Y" e up="Z" deixam o arquivo igual à cena do Blender (mesma convenção do export_blender_mesh.py).
"""
import os
import sys

import bpy
from mathutils import Matrix

ROOT = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
OUT = sys.argv[sys.argv.index("--") + 1] if "--" in sys.argv else os.path.join(
    ROOT, "robot", "ros2_ws", "src", "geraldo_description", "meshes", "hospital"
)
PARTS = {
    "hospital_estrutura": "HOSPITAL_estrutura",
    "hospital_mobiliario": "HOSPITAL_mobiliario",
    "hospital_obstaculos": "HOSPITAL_obstaculos",
    "maca": "HOSPITAL_maca",
}

os.makedirs(OUT, exist_ok=True)
for name, collection in PARTS.items():
    meshes = [o for o in bpy.data.collections[collection].objects if o.type == "MESH"]
    assert meshes, f"nenhuma malha em {collection}"
    saved = {o: o.matrix_world.copy() for o in meshes} if name == "maca" else {}
    for o in saved:
        o.matrix_world = Matrix.Identity(4)
    bpy.ops.object.select_all(action="DESELECT")
    for o in meshes:
        o.select_set(True)
    bpy.ops.wm.obj_export(
        filepath=os.path.join(OUT, f"{name}.obj"),
        export_selected_objects=True,
        apply_modifiers=True,
        export_materials=True,
        forward_axis="Y",
        up_axis="Z",
        global_scale=1.0,
    )
    for o, m in saved.items():
        o.matrix_world = m
    print("EXPORT_OK", name, len(meshes))
```

# Coloca (on, padrão) ou tira (off) a maca que bloqueia o corredor norte, com a simulação rodando.
