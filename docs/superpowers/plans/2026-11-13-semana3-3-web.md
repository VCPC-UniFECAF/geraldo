## Decisões deste plano (o coordenador confere a consistência com os outros planos)

1. **Autenticação do WebSocket:** o navegador não envia cabeçalho `Authorization` num WebSocket, então o web abre `WS /ws/telemetry?token=<JWT>`. **O plano do backend precisa aceitar o JWT no parâmetro `token` da query** (e fechar a conexão com código 1008 se faltar ou for inválido).
2. **Estado da parada de emergência:** o contrato só deixa o papel `robo` ler `GET /robot/estop`; o web não lê. A tela mostra "ativa" quando acabou de acionar pelo botão (`solicitado`) ou, sem ação local, quando a telemetria traz `robot_state = PARADA_EMERGENCIA`.
3. **Navegação:** 4 abas (limite do design system para a barra inferior): Solicitar, Minhas entregas, Estoque (com abas internas Saldos e Histórico) e **Painel** (robô ao vivo, fila, entregas em andamento, botão de emergência).
4. **Tempo estimado do `RobotStatusCard`:** não existe no contrato; é calculado do comprimento restante da `route` a 3,0 m/s (decisão de velocidade do spec) e exibido como "Cerca de N s".
5. **Mock de demonstração:** `npm run dev:demo` ganha usuários `medico1`, `farmaceutico1` e `admin` (senha `demo`) e um robô simulado que percorre todos os estados com os corpos de `docs/contracts/examples/robot-telemetry.json`.

```ts
import { ApiError, type ApiClient } from "./ApiClient";
import type { Delivery, DeliveryCreate, EstopState, Location, Me, Movement, MovementFiltro, StockEntry, Telemetry } from "./types";

export class HttpApiClient implements ApiClient {
  #token: string | null = null;
  readonly #baseUrl: string;

  constructor(baseUrl: string = import.meta.env.VITE_API_URL ?? "http://localhost:8000") {
    this.#baseUrl = baseUrl;
  }

  definirToken(token: string | null): void {
    this.#token = token;
  }

  async #request<T>(path: string, init: RequestInit = {}): Promise<T> {
    const headers = new Headers(init.headers);
    if (init.body) headers.set("Content-Type", "application/json");
    if (this.#token) headers.set("Authorization", `Bearer ${this.#token}`);
    const res = await fetch(`${this.#baseUrl}${path}`, { ...init, headers });
    if (!res.ok) {
      const body = await res.json().catch(() => ({}));
      throw new ApiError(res.status, body.detail ?? "Algo deu errado. Tente de novo.");
    }
    return (res.status === 204 ? undefined : await res.json()) as T;
  }

  login(username: string, password: string) {
    return this.#request<{ access_token: string; token_type: "bearer" }>("/auth/login", {
      method: "POST",
      body: JSON.stringify({ username, password }),
    });
  }

  me() {
    return this.#request<Me>("/me");
  }

  locations() {
    return this.#request<Location[]>("/locations");
  }

  stock(locationId?: string) {
    return this.#request<StockEntry[]>(`/stock${locationId ? `?location_id=${encodeURIComponent(locationId)}` : ""}`);
  }

  createDelivery(body: DeliveryCreate) {
    return this.#request<Delivery>("/deliveries", { method: "POST", body: JSON.stringify(body) });
  }

  deliveries() {
    return this.#request<Delivery[]>("/deliveries");
  }

  confirmReceipt(id: number) {
    return this.#request<Delivery>(`/deliveries/${id}/confirm-receipt`, { method: "POST" });
  }

  movements(filtro: MovementFiltro = {}) {
    const query = new URLSearchParams();
    if (filtro.deliveryId !== undefined) query.set("delivery_id", String(filtro.deliveryId));
    if (filtro.locationId) query.set("location_id", filtro.locationId);
    const texto = query.toString();
    return this.#request<Movement[]>(`/movements${texto ? `?${texto}` : ""}`);
  }

  telemetry() {
    return this.#request<Telemetry>("/telemetry");
  }

  setEstop(active: boolean) {
    const corpo: EstopState = { active };
    return this.#request<void>("/robot/estop", { method: "POST", body: JSON.stringify(corpo) });
  }

  telemetryStreamUrl() {
    const url = `${this.#baseUrl.replace(/^http/, "ws")}/ws/telemetry`;
    return this.#token ? `${url}?token=${encodeURIComponent(this.#token)}` : url;
  }
}
```

```ts
import type { RobotState, Telemetry } from "../types";
import { TELEMETRIAS } from "./fixtures";

// Roteiro do modo demo: o robô passa pelos estados de uma entrega, 4 avanços (4 s) em cada.
const ROTEIRO: RobotState[] = ["IDLE", "CARREGANDO", "A_CAMINHO", "CHEGOU", "AGUARDANDO_RETIRADA", "VOLTANDO"];
const AVANCOS_POR_ESTADO = 4;

// Robô de mentira para o dev e os testes: a parada de emergência vence o roteiro.
export class RoboSimulado {
  #estop = false;
  #passo = 0;

  definirEstop(ativo: boolean): void {
    this.#estop = ativo;
  }

  reiniciar(): void {
    this.#estop = false;
    this.#passo = 0;
  }

  atual(): Telemetry {
    const estado: RobotState = this.#estop
      ? "PARADA_EMERGENCIA"
      : ROTEIRO[Math.floor(this.#passo / AVANCOS_POR_ESTADO) % ROTEIRO.length];
    return { ...TELEMETRIAS[estado], updated_at: new Date().toISOString() };
  }

  avancar(): Telemetry {
    this.#passo += 1;
    return this.atual();
  }
}

export const roboSimulado = new RoboSimulado();
```

```tsx
import { PackageCheck } from "lucide-react";
import { useCallback, useEffect, useState } from "react";
import type { CatalogoApi, EntregasApi } from "../api/ApiClient";
import { Button } from "../components/Button";
import { Card } from "../components/Card";
import { DeliveryQueueRow } from "../components/DeliveryQueueRow";
import { EntregaAdapter, type Entrega } from "../domain/EntregaAdapter";
import { aguardandoRecebimento, maisRecentesPrimeiro } from "../domain/operacoes";

interface Props {
  api: CatalogoApi & EntregasApi;
  onRequest: () => void;
}

export function MinhasEntregasPage({ api, onRequest }: Props) {
  const [entregas, setEntregas] = useState<Entrega[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [confirmando, setConfirmando] = useState<number | null>(null);
  const [erroAcao, setErroAcao] = useState<string | null>(null);
  const [confirmada, setConfirmada] = useState(false);

  const carregar = useCallback(async () => {
    const [deliveries, locations] = await Promise.all([api.deliveries(), api.locations()]);
    const adapter = new EntregaAdapter((id) => locations.find((l) => l.id === id)?.name ?? id);
    setEntregas(maisRecentesPrimeiro(deliveries).map((d) => adapter.adaptar(d)));
  }, [api]);

  useEffect(() => {
    carregar().catch((err: Error) => setError(err.message));
  }, [carregar]);

  async function confirmar(id: number) {
    setConfirmando(id);
    setErroAcao(null);
    setConfirmada(false);
    try {
      await api.confirmReceipt(id);
      await carregar();
      setConfirmada(true);
    } catch (err) {
      setErroAcao(err instanceof Error ? err.message : "Não foi possível confirmar o recebimento. Tente de novo.");
    } finally {
      setConfirmando(null);
    }
  }

  if (error) return <p role="alert" className="text-vermelho-600">{error}</p>;
  if (!entregas) return <p className="text-neutro-500">Carregando…</p>;

  return (
    <section>
      <h1 className="font-display text-2xl font-bold">Minhas entregas</h1>
      {confirmada && (
        <p role="status" className="mt-4 rounded-controle bg-sucesso-50 px-4 py-3 text-sucesso-700">
          Recebimento confirmado. A tampa do robô foi liberada.
        </p>
      )}
      {erroAcao && (
        <p role="alert" className="mt-4 rounded-controle border border-vermelho-200 bg-vermelho-50 px-3 py-2 text-sm text-vermelho-600">
          {erroAcao}
        </p>
      )}
      <Card className="mt-4">
        {entregas.length === 0 ? (
          <div className="grid justify-items-center gap-3 py-6 text-center">
            <PackageCheck size={40} aria-hidden="true" className="text-neutro-300" />
            <p className="font-display text-lg font-semibold">Nenhuma entrega ainda</p>
            <p className="text-sm text-neutro-500">Solicite um medicamento para acompanhar a entrega aqui.</p>
            <Button onClick={onRequest}>Solicitar medicamento</Button>
          </div>
        ) : (
          <ul className="grid gap-2">
            {entregas.map((e) => (
              <DeliveryQueueRow
                key={e.id}
                entrega={e}
                acao={
                  aguardandoRecebimento(e) ? (
                    <Button
                      variant="secundario"
                      className="w-full md:w-auto"
                      disabled={confirmando === e.id}
                      aria-label={`Confirmar recebimento da entrega ${e.id}`}
                      onClick={() => confirmar(e.id)}
                    >
                      <PackageCheck size={20} aria-hidden="true" />
                      Confirmar recebimento
                    </Button>
                  ) : undefined
                }
              />
            ))}
          </ul>
        )}
      </Card>
    </section>
  );
}
```

```ts
import { CircleAlert, TriangleAlert, type LucideIcon } from "lucide-react";

export type NivelId = "ok" | "atencao" | "abaixo";

// STRATEGY: cada nível sabe sua cor, seu texto e seu ícone (DS §3.5). Novo nível = nova classe (OCP).
export abstract class NivelEstoqueStrategy {
  abstract readonly valor: NivelId;
  abstract readonly barra: string;
  abstract readonly texto: string;
  abstract readonly rotulo: string | null;
  abstract readonly Icone: LucideIcon | null;
}

class Normal extends NivelEstoqueStrategy {
  readonly valor = "ok";
  readonly barra = "bg-azul-500";
  readonly texto = "text-neutro-600";
  readonly rotulo = null;
  readonly Icone = null;
}

class PertoDoMinimo extends NivelEstoqueStrategy {
  readonly valor = "atencao";
  readonly barra = "bg-aviso-500";
  readonly texto = "text-aviso-700";
  readonly rotulo = "Perto do mínimo";
  readonly Icone = CircleAlert;
}

class AbaixoDoMinimo extends NivelEstoqueStrategy {
  readonly valor = "abaixo";
  readonly barra = "bg-vermelho-500";
  readonly texto = "text-vermelho-600";
  readonly rotulo = "Abaixo do mínimo";
  readonly Icone = TriangleAlert;
}

const NORMAL = new Normal();
const PERTO = new PertoDoMinimo();
const ABAIXO = new AbaixoDoMinimo();

// FACTORY: acima de 120% do mínimo = normal; de 100% a 120% = perto; abaixo do mínimo = abaixo.
export function nivelDe(quantidade: number, minimo: number): NivelEstoqueStrategy {
  if (quantidade < minimo) return ABAIXO;
  if (quantidade <= minimo * 1.2) return PERTO;
  return NORMAL;
}

// A barra enche até o dobro do mínimo (o mínimo fica no meio); sem mínimo, fica cheia.
export function preenchimento(quantidade: number, minimo: number): number {
  if (minimo <= 0) return 100;
  return Math.min(100, Math.round((quantidade / (minimo * 2)) * 100));
}
```

```tsx
import { useEffect, useState } from "react";
import type { CatalogoApi } from "../../api/ApiClient";
import type { Location, StockEntry } from "../../api/types";
import { AlertBanner } from "../../components/AlertBanner";
import { StockLevelBar } from "../../components/StockLevelBar";
import { abaixoDoMinimo, abaixoDoMinimoPrimeiro, locaisComEstoque } from "../../domain/operacoes";

const FIELD =
  "min-h-controle w-full rounded-controle border border-neutro-200 bg-branco px-3 text-base text-tinta " +
  "focus:border-azul-500 focus:outline-none focus:ring-2 focus:ring-azul-200";

export function SaldosPanel({ api }: { api: CatalogoApi }) {
  const [locais, setLocais] = useState<Location[]>([]);
  const [localId, setLocalId] = useState("");
  const [estoque, setEstoque] = useState<StockEntry[] | null>(null);
  const [erro, setErro] = useState<string | null>(null);

  useEffect(() => {
    api.locations().then(setLocais).catch((err: Error) => setErro(err.message));
  }, [api]);

  useEffect(() => {
    let cancelado = false;
    setEstoque(null);
    setErro(null);
    api
      .stock(localId || undefined)
      .then((entradas) => {
        if (!cancelado) setEstoque(entradas);
      })
      .catch((err: Error) => {
        if (!cancelado) setErro(err.message);
      });
    return () => {
      cancelado = true;
    };
  }, [api, localId]);

  const nomeDoLocal = (id: string) => locais.find((l) => l.id === id)?.name ?? id;
  const faltando = estoque ? abaixoDoMinimo(estoque).length : 0;

  return (
    <div className="grid gap-4">
      <div className="md:max-w-xs">
        <label htmlFor="local" className="mb-1 block text-sm font-medium text-neutro-700">
          Local
        </label>
        <select id="local" className={FIELD} value={localId} onChange={(e) => setLocalId(e.target.value)}>
          <option value="">Todos os locais</option>
          {locaisComEstoque(locais).map((l) => (
            <option key={l.id} value={l.id}>
              {l.name}
            </option>
          ))}
        </select>
      </div>
      {erro && (
        <p role="alert" className="rounded-controle border border-vermelho-200 bg-vermelho-50 px-3 py-2 text-sm text-vermelho-600">
          {erro}
        </p>
      )}
      {faltando > 0 && (
        <AlertBanner>
          {faltando} {faltando === 1 ? "item abaixo do mínimo" : "itens abaixo do mínimo"}. Peça a reposição à farmácia.
        </AlertBanner>
      )}
      {estoque === null ? (
        !erro && <p className="text-neutro-500">Carregando…</p>
      ) : estoque.length === 0 ? (
        <p className="text-sm text-neutro-500">Nenhum item neste local.</p>
      ) : (
        <ul className="grid gap-2 md:grid-cols-2">
          {abaixoDoMinimoPrimeiro(estoque).map((e) => (
            <StockLevelBar key={`${e.location_id}/${e.item_id}`} entrada={e} local={nomeDoLocal(e.location_id)} />
          ))}
        </ul>
      )}
    </div>
  );
}
```

```ts
// PROGRAMAÇÃO FUNCIONAL: funções puras sobre o corpo de telemetria do contrato.
import { ROBOT_STATES, type Ponto, type RobotState, type Telemetry } from "../api/types";

// Velocidade do robô decidida na semana 3 (3,0 m/s sempre); só serve para a estimativa de chegada.
export const VELOCIDADE_DO_ROBO = 3.0;

const LIMITE_DA_BATERIA_BAIXA = 20;

// Valida o texto de uma mensagem do WebSocket. Devolve null se não for uma telemetria do contrato.
export function lerTelemetria(texto: string): Telemetry | null {
  let corpo: unknown;
  try {
    corpo = JSON.parse(texto);
  } catch {
    return null;
  }
  if (typeof corpo !== "object" || corpo === null) return null;
  const c = corpo as Partial<Telemetry>;
  const pose = c.pose;
  if (!ROBOT_STATES.includes(c.robot_state as RobotState)) return null;
  if (!pose || typeof pose.x !== "number" || typeof pose.y !== "number" || typeof pose.yaw !== "number") return null;
  if (typeof c.battery !== "number") return null;
  return {
    robot_state: c.robot_state as RobotState,
    pose,
    battery: c.battery,
    delivery_id: c.delivery_id ?? null,
    route: Array.isArray(c.route) ? c.route : [],
    obstacles: Array.isArray(c.obstacles) ? c.obstacles : [],
    updated_at: String(c.updated_at ?? ""),
  };
}

// Espera antes de reconectar: 1 s, 2 s, 4 s, 8 s e no máximo 10 s.
export const atrasoDeReconexao = (tentativa: number): number => Math.min(1000 * 2 ** tentativa, 10000);

export const comprimentoDaRota = (rota: Ponto[]): number =>
  rota.slice(1).reduce((total, ponto, i) => total + Math.hypot(ponto.x - rota[i].x, ponto.y - rota[i].y), 0);

export function segundosAteChegar(rota: Ponto[] | undefined, velocidade: number = VELOCIDADE_DO_ROBO): number | null {
  const metros = comprimentoDaRota(rota ?? []);
  return metros > 0 ? Math.ceil(metros / velocidade) : null;
}

export const bateriaBaixa = (bateria: number): boolean => bateria < LIMITE_DA_BATERIA_BAIXA;
```

```ts
// O pedaço de WebSocket que usamos: o teste troca por um socket falso (src/test/SocketFalso.ts).
export interface SocketLike {
  onopen: ((evento: unknown) => void) | null;
  onmessage: ((evento: { data: unknown }) => void) | null;
  onclose: ((evento: unknown) => void) | null;
  onerror: ((evento: unknown) => void) | null;
  close(): void;
}

export type EstadoConexao = "conectando" | "conectado" | "reconectando" | "parado";

export type EventoTelemetria =
  | { tipo: "telemetria"; dados: Telemetry }
  | { tipo: "conexao"; estado: EstadoConexao };

// OBSERVER: sujeito que mantém o WebSocket /ws/telemetry aberto (reconecta sozinho) e avisa os observers.
// A URL é uma função porque o token pode mudar entre uma tentativa e outra (ApiClient.telemetryStreamUrl).
export class TelemetriaCliente extends Observavel<EventoTelemetria> {
  readonly #urlDoStream: () => string;
  readonly #criarSocket: (url: string) => SocketLike;
  #socket: SocketLike | null = null;
  #timer: ReturnType<typeof setTimeout> | null = null;
  #tentativas = 0;
  #ativo = false;

  constructor(
    urlDoStream: () => string,
    criarSocket: (url: string) => SocketLike = (url) => new WebSocket(url) as unknown as SocketLike,
  ) {
    super();
    this.#urlDoStream = urlDoStream;
    this.#criarSocket = criarSocket;
  }

  iniciar(): void {
    if (this.#ativo) return;
    this.#ativo = true;
    this.#tentativas = 0;
    this.notificar({ tipo: "conexao", estado: "conectando" });
    this.#conectar();
  }

  parar(): void {
    this.#ativo = false;
    if (this.#timer !== null) {
      clearTimeout(this.#timer);
      this.#timer = null;
    }
    const socket = this.#socket;
    this.#socket = null;
    if (socket) {
      socket.onopen = null;
      socket.onmessage = null;
      socket.onclose = null;
      socket.onerror = null;
      socket.close();
    }
    this.notificar({ tipo: "conexao", estado: "parado" });
  }

  #conectar(): void {
    let socket: SocketLike;
    try {
      socket = this.#criarSocket(this.#urlDoStream());
    } catch {
      this.#reagendar();
      return;
    }
    this.#socket = socket;
    socket.onopen = () => {
      this.#tentativas = 0;
      this.notificar({ tipo: "conexao", estado: "conectado" });
    };
    socket.onmessage = (evento) => {
      const dados = lerTelemetria(String(evento.data));
      if (dados) this.notificar({ tipo: "telemetria", dados });
    };
    socket.onclose = () => {
      if (this.#socket !== socket) return;
      this.#socket = null;
      this.#reagendar();
    };
  }

  #reagendar(): void {
    const espera = atrasoDeReconexao(this.#tentativas);
    this.#tentativas += 1;
    this.notificar({ tipo: "conexao", estado: "reconectando" });
    this.#timer = setTimeout(() => {
      this.#timer = null;
      this.#conectar();
    }, espera);
  }
}
```

```ts
  }

    this.onopen?.({});
  }

  receber(corpo: unknown): void {
    this.onmessage?.({ data: typeof corpo === "string" ? corpo : JSON.stringify(corpo) });
  }

  cair(): void {
    this.onclose?.({});
  }
}

  const criar = (url: string): SocketFalso => {
    const socket = new SocketFalso(url);
    sockets.push(socket);
    return socket;
  };
```

`web/src/domain/mapa.ts`:

```ts
import type { Ponto, Telemetry } from "../api/types";

export interface Vista {
  robo: Ponto & { yaw: number };
  rota: Ponto[];
  obstaculos: { x: number; y: number; raio: number }[];
}

const ESCALA_MAXIMA = 40; // px por metro: um robô sozinho não vira um ponto gigante
const RAIO_MINIMO = 3; // px: obstáculo pequeno continua visível

// Função pura: leva metros do mapa para pixels do SVG (no mapa o y cresce para cima; no SVG, para baixo).
export function enquadrar(t: Telemetry, largura: number, altura: number, margem = 16): Vista {
  const rota = t.route ?? [];
  const obstaculos = t.obstacles ?? [];
  const xs = [t.pose.x, ...rota.map((p) => p.x), ...obstaculos.flatMap((o) => [o.x - o.radius, o.x + o.radius])];
  const ys = [t.pose.y, ...rota.map((p) => p.y), ...obstaculos.flatMap((o) => [o.y - o.radius, o.y + o.radius])];
  const minX = Math.min(...xs);
  const minY = Math.min(...ys);
  const larguraUtil = largura - 2 * margem;
  const alturaUtil = altura - 2 * margem;
  const larguraReal = Math.max(...xs) - minX;
  const alturaReal = Math.max(...ys) - minY;
  const escala = Math.min(larguraUtil / Math.max(larguraReal, 1), alturaUtil / Math.max(alturaReal, 1), ESCALA_MAXIMA);
  const folgaX = (larguraUtil - larguraReal * escala) / 2;
  const folgaY = (alturaUtil - alturaReal * escala) / 2;
  const px = (x: number) => margem + folgaX + (x - minX) * escala;
  const py = (y: number) => altura - (margem + folgaY + (y - minY) * escala);
  return {
    robo: { x: px(t.pose.x), y: py(t.pose.y), yaw: t.pose.yaw },
    rota: rota.map((p) => ({ x: px(p.x), y: py(p.y) })),
    obstaculos: obstaculos.map((o) => ({ x: px(o.x), y: py(o.y), raio: Math.max(o.radius * escala, RAIO_MINIMO) })),
  };
}
```

```ts
import { useEffect, useState } from "react";
import type { Telemetry } from "../api/types";
import type { Observador } from "../domain/Observavel";
import type { EstadoConexao, EventoTelemetria, TelemetriaCliente } from "../domain/TelemetriaCliente";

// Liga o TelemetriaCliente (Observer) ao React. `snapshot` busca o primeiro estado (GET /telemetry) enquanto o
// WebSocket conecta; se o WebSocket responder antes, o dado dele vale mais. `snapshot` deve ser estável (useCallback).
export function useTelemetria(cliente: TelemetriaCliente, snapshot: () => Promise<Telemetry>) {
  const [telemetria, setTelemetria] = useState<Telemetry | null>(null);
  const [conexao, setConexao] = useState<EstadoConexao>("parado");

  useEffect(() => {
    let ativo = true;
    const observer: Observador<EventoTelemetria> = {
      atualizar: (evento) => {
        if (evento.tipo === "telemetria") setTelemetria(evento.dados);
        else setConexao(evento.estado);
      },
    };
    cliente.adicionarObserver(observer);
    snapshot()
      .then((inicial) => {
        if (ativo) setTelemetria((atual) => atual ?? inicial);
      })
      .catch(() => {}); // sem snapshot, o painel espera o WebSocket
    cliente.iniciar();
    return () => {
      ativo = false;
      cliente.removerObserver(observer);
      cliente.parar();
    };
  }, [cliente, snapshot]);

  return { telemetria, conexao };
}
```

```tsx
  telemetria: TelemetriaCliente;

function Secao({ titulo, entregas, vazio }: { titulo: string; entregas: Entrega[]; vazio: string }) {
  return (
    <Card role="region" aria-label={titulo}>
      <h2 className="font-display text-lg font-semibold">{titulo}</h2>
      {entregas.length === 0 ? (
        <p className="mt-3 text-sm text-neutro-500">{vazio}</p>
      ) : (
        <ul className="mt-3 grid gap-2">
          {entregas.map((e) => (
            <DeliveryQueueRow key={e.id} entrega={e} />
          ))}
        </ul>
      )}
    </Card>
  );

export function PainelPage({ api, telemetria: cliente }: Props) {
  const snapshot = useCallback(() => api.telemetry(), [api]);
  const { telemetria, conexao } = useTelemetria(cliente, snapshot);
  const [dados, setDados] = useState<{ entregas: Delivery[]; locais: Location[] } | null>(null);
  const [erro, setErro] = useState<string | null>(null);

  const estado = telemetria?.robot_state;
  const entregaId = telemetria?.delivery_id ?? null;

  // Recarrega a fila quando o robô muda de estado ou de entrega (é o que muda a fila de verdade).
  useEffect(() => {
    let ativo = true;
    Promise.all([api.deliveries(), api.locations()])
      .then(([entregas, locais]) => {
        if (ativo) setDados({ entregas, locais });
      })
      .catch((err: Error) => {
        if (ativo) setErro(err.message);
      });
    return () => {
      ativo = false;
    };
  }, [api, estado, entregaId]);

  const nomeDoLocal = (id: string) => dados?.locais.find((l) => l.id === id)?.name ?? id;
  const adapter = new EntregaAdapter(nomeDoLocal);
  const adaptar = (lista: Delivery[]) => lista.map((d) => adapter.adaptar(d));
  const emCurso = dados?.entregas.find((e) => e.id === entregaId);
  const entrega = emCurso ? `Entrega #${emCurso.id} · ${nomeDoLocal(emCurso.dest_id)}` : entregaId !== null ? `Entrega #${entregaId}` : null;
  const emergenciaAtiva = telemetria ? estadoRoboDe(telemetria.robot_state).emEmergencia : false;

  return (
    <section className="grid gap-4">
      <h1 className="font-display text-2xl font-bold">Painel do robô</h1>
      {emergenciaAtiva && <AlertBanner>Parada de emergência ativa. O robô está parado até a liberação.</AlertBanner>}
      {telemetria && bateriaBaixa(telemetria.battery) && <AlertBanner>Bateria abaixo de 20%. Evite novos pedidos até a recarga.</AlertBanner>}
      {conexao === "reconectando" && <AlertBanner variante="info">Sem conexão com o robô. Tentando reconectar…</AlertBanner>}
      {erro && (
        <p role="alert" className="rounded-controle border border-vermelho-200 bg-vermelho-50 px-3 py-2 text-sm text-vermelho-600">
          {erro}
        </p>
      )}
      <div className="grid gap-4 md:grid-cols-2">
        <RobotStatusCard telemetria={telemetria} entrega={entrega} />
        <TelemetryPanel telemetria={telemetria} />
      </div>
      <Secao titulo="Em andamento" entregas={dados ? adaptar(emAndamento(dados.entregas)) : []} vazio="Nenhuma entrega em andamento." />
      <Secao titulo="Fila de espera" entregas={dados ? adaptar(filaDeEspera(dados.entregas)) : []} vazio="Ninguém na fila." />
```

```tsx
import type { CatalogoApi, EntregasApi, RoboApi } from "../api/ApiClient";
import type { Delivery, Location, Role } from "../api/types";
import { AlertBanner } from "../components/AlertBanner";
import { BotaoEmergencia } from "../components/BotaoEmergencia";
import { Card } from "../components/Card";
import { DeliveryQueueRow } from "../components/DeliveryQueueRow";
import { RobotStatusCard } from "../components/RobotStatusCard";
import { TelemetryPanel } from "../components/TelemetryPanel";
import { EntregaAdapter, type Entrega } from "../domain/EntregaAdapter";
import { emAndamento, filaDeEspera } from "../domain/operacoes";
import { bateriaBaixa } from "../domain/telemetria";
import type { TelemetriaCliente } from "../domain/TelemetriaCliente";
import { useTelemetria } from "../hooks/useTelemetria";

interface Props {
  api: CatalogoApi & EntregasApi & RoboApi;
  telemetria: TelemetriaCliente;
  role: Role;
}

function Secao({ titulo, entregas, vazio }: { titulo: string; entregas: Entrega[]; vazio: string }) {
  return (
    <Card role="region" aria-label={titulo}>
      <h2 className="font-display text-lg font-semibold">{titulo}</h2>
      {entregas.length === 0 ? (
        <p className="mt-3 text-sm text-neutro-500">{vazio}</p>
      ) : (
        <ul className="mt-3 grid gap-2">
          {entregas.map((e) => (
            <DeliveryQueueRow key={e.id} entrega={e} />
          ))}
        </ul>
      )}
    </Card>
  );
}

export function PainelPage({ api, telemetria: cliente, role }: Props) {
  const snapshot = useCallback(() => api.telemetry(), [api]);
  const { telemetria, conexao } = useTelemetria(cliente, snapshot);
  const [dados, setDados] = useState<{ entregas: Delivery[]; locais: Location[] } | null>(null);
  const [erro, setErro] = useState<string | null>(null);
  // `solicitado` é o que este painel acabou de pedir; null = sem pedido pendente, vale o que a telemetria diz.
  const [solicitado, setSolicitado] = useState<boolean | null>(null);
  const [ocupado, setOcupado] = useState(false);
  const [erroEstop, setErroEstop] = useState<string | null>(null);

  const estado = telemetria?.robot_state;
  const entregaId = telemetria?.delivery_id ?? null;
  const robotParado = estado === "PARADA_EMERGENCIA";
  const emergenciaAtiva = solicitado ?? robotParado;

  // Recarrega a fila quando o robô muda de estado ou de entrega (é o que muda a fila de verdade).
  useEffect(() => {
    let ativo = true;
    Promise.all([api.deliveries(), api.locations()])
      .then(([entregas, locais]) => {
        if (ativo) setDados({ entregas, locais });
      })
      .catch((err: Error) => {
        if (ativo) setErro(err.message);
      });
    return () => {
      ativo = false;
    };
  }, [api, estado, entregaId]);

  // Quando a telemetria confirma o que pedimos, o pedido local deixa de ser necessário.
  useEffect(() => {
    if (solicitado !== null && telemetria && robotParado === solicitado) setSolicitado(null);
  }, [solicitado, telemetria, robotParado]);

  async function alternarEmergencia(ativar: boolean) {
    setOcupado(true);
    setErroEstop(null);
    try {
      await api.setEstop(ativar);
      setSolicitado(ativar);
    } catch (err) {
      setErroEstop(err instanceof Error ? err.message : "Não foi possível acionar a parada de emergência. Tente de novo.");
    } finally {
      setOcupado(false);
    }
  }

  const nomeDoLocal = (id: string) => dados?.locais.find((l) => l.id === id)?.name ?? id;
  const adapter = new EntregaAdapter(nomeDoLocal);
  const adaptar = (lista: Delivery[]) => lista.map((d) => adapter.adaptar(d));
  const emCurso = dados?.entregas.find((e) => e.id === entregaId);
  const entrega = emCurso ? `Entrega #${emCurso.id} · ${nomeDoLocal(emCurso.dest_id)}` : entregaId !== null ? `Entrega #${entregaId}` : null;

  return (
    <section className="grid gap-4">
      <h1 className="font-display text-2xl font-bold">Painel do robô</h1>
      {emergenciaAtiva && <AlertBanner>Parada de emergência ativa. O robô está parado até a liberação.</AlertBanner>}
      {telemetria && bateriaBaixa(telemetria.battery) && <AlertBanner>Bateria abaixo de 20%. Evite novos pedidos até a recarga.</AlertBanner>}
      {conexao === "reconectando" && <AlertBanner variante="info">Sem conexão com o robô. Tentando reconectar…</AlertBanner>}
      {erro && (
        <p role="alert" className="rounded-controle border border-vermelho-200 bg-vermelho-50 px-3 py-2 text-sm text-vermelho-600">
          {erro}
        </p>
      )}
      <BotaoEmergencia role={role} ativo={emergenciaAtiva} ocupado={ocupado} onAlternar={alternarEmergencia} />
      {erroEstop && (
        <p role="alert" className="rounded-controle border border-vermelho-200 bg-vermelho-50 px-3 py-2 text-sm text-vermelho-600">
          {erroEstop}
        </p>
      )}
      <div className="grid gap-4 md:grid-cols-2">
        <RobotStatusCard telemetria={telemetria} entrega={entrega} />
        <TelemetryPanel telemetria={telemetria} />
      </div>
      <Secao titulo="Em andamento" entregas={dados ? adaptar(emAndamento(dados.entregas)) : []} vazio="Nenhuma entrega em andamento." />
      <Secao titulo="Fila de espera" entregas={dados ? adaptar(filaDeEspera(dados.entregas)) : []} vazio="Ninguém na fila." />
    </section>
  );
}
```

```tsx
import { useEffect, useMemo, useState } from "react";
import type { ApiClient } from "./api/ApiClient";
import { HttpApiClient } from "./api/HttpApiClient";
import type { Me } from "./api/types";
import { AppShell, type PageId } from "./components/AppShell";
import { Sessao, TokenObserver } from "./domain/Sessao";
import { TelemetriaCliente } from "./domain/TelemetriaCliente";
import { EstoquePage } from "./pages/EstoquePage";
import { LoginPage } from "./pages/LoginPage";
import { MinhasEntregasPage } from "./pages/MinhasEntregasPage";
import { PainelPage } from "./pages/PainelPage";
import { SolicitarPage } from "./pages/SolicitarPage";

// DIP: quem monta o app escolhe a implementação da API e do cliente de telemetria; as telas só conhecem as interfaces.
export default function App({ api, telemetria }: { api?: ApiClient; telemetria?: TelemetriaCliente }) {
  const cliente = useMemo<ApiClient>(() => api ?? new HttpApiClient(), [api]);
  const stream = useMemo(() => telemetria ?? new TelemetriaCliente(() => cliente.telemetryStreamUrl()), [telemetria, cliente]);
  const sessao = useMemo(() => {
    const s = new Sessao(cliente);
    s.adicionarObserver(new TokenObserver(cliente)); // primeiro o token, depois a tela
    return s;
  }, [cliente]);
  const [logado, setLogado] = useState(false);
  const [me, setMe] = useState<Me | null>(null);
  const [erroMe, setErroMe] = useState<string | null>(null);
  const [page, setPage] = useState<PageId>("solicitar");

  useEffect(() => {
    const observer = { atualizar: () => setLogado(true) };
    sessao.adicionarObserver(observer);
    return () => sessao.removerObserver(observer);
  }, [sessao]);

  // GET /me: o papel decide o que aparece (botão de emergência).
  useEffect(() => {
    if (!logado) return;
    cliente.me().then(setMe).catch((err: Error) => setErroMe(err.message));
  }, [logado, cliente]);

  if (!logado) return <LoginPage sessao={sessao} />;
  if (erroMe) return <p role="alert" className="p-4 text-vermelho-600">{erroMe}</p>;
  if (!me) return <p className="p-4 text-neutro-500">Carregando…</p>;
  return (
    <AppShell active={page} onNavigate={setPage}>
```
